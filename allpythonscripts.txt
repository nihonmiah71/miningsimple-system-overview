CONFIG_FILE_NAME = "config.txt"


# ==============================================================================
# CONFIG LOADER
# ==============================================================================

def load_config_data(base_dir):
    """Loads configuration data from config.txt."""
    cfg_path = os.path.join(base_dir, CONFIG_FILE_NAME)
    if not os.path.exists(cfg_path):
        # Check current working directory as fallback
        cfg_path = os.path.join(os.getcwd(), CONFIG_FILE_NAME)

    if not os.path.exists(cfg_path):
        return None, cfg_path

    try:
        with open(cfg_path, "r", encoding="utf-8") as f:
            return json.load(f), cfg_path
    except Exception as e:
        print(f"Error loading {cfg_path}: {e}")
        return None, cfg_path


# ==============================================================================

class RefactorEngine:
    def __init__(self, config_data, anki_addons_dir, repo_dir, logger=print):
        self.cfg = config_data
        self.addons_dir = anki_addons_dir
        self.repo_dir = repo_dir
        self.log = logger
        
        self.decks = self.cfg.get("decks", {})
        self.models = self.cfg.get("models", {})
        self.templates = self.cfg.get("templates", {})
        self.m_fields = self.cfg.get("mining_fields", {})
        self.g_fields = self.cfg.get("grammar_fields", {})

    def get_val(self, group, key, default=None):
        if default is None:
            default = key
        return self.cfg.get(group, {}).get(key, default)

    def replace_in_file(self, file_path, replacer_callback):
        """Safely updates a file in place using a callback function."""
        if not os.path.exists(file_path):
            self.log(f"[SKIPPED - Not Found] {file_path}")
            return False

        try:
            with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
                content = f.read()

            new_content = replacer_callback(content)

            if new_content != content:
                with open(file_path, "w", encoding="utf-8") as f:
                    f.write(new_content)
                self.log(f"[UPDATED] {file_path}")
                return True
            else:
                self.log(f"[UNCHANGED] {file_path}")
                return False
        except Exception as e:
            self.log(f"[ERROR] Failed modifying {file_path}: {e}")
            return False

    def find_addon_file(self, addon_id, filename):
        """Locates an add-on file either inside a folder named addon_id or subfolders."""
        if not self.addons_dir or not os.path.exists(self.addons_dir):
            return None
        
        # Check direct folder
        direct_path = os.path.join(self.addons_dir, str(addon_id), filename)
        if os.path.exists(direct_path):
            return direct_path

        # Scan folder names containing the ID
        for root, dirs, files in os.walk(self.addons_dir):
            if os.path.basename(root) == str(addon_id) or str(addon_id) in os.path.basename(root):
                if filename in files:
                    return os.path.join(root, filename)
                # Check init alternative
                if filename == "__init__.py" and "init.py" in files:
                    return os.path.join(root, "init.py")

        return direct_path

    def refactor_addon_207985417(self):
        """field_extract_inject_config.json"""
        fpath = self.find_addon_file("207985417", "field_extract_inject_config.json")
        if not os.path.exists(fpath):
            fpath = self.find_addon_file("207985417", "config.json")

        def transform(text):
            try:
                data = json.loads(text)
            except Exception:
                return text

            m_model = self.get_val("models", "miningsimple")
            g_model = self.get_val("models", "Grammar")

            # Update extraction profiles
            for prof_name, prof in data.get("profiles", {}).get("extraction", {}).items():
                if prof.get("model_name") in ["miningsimple", m_model]:
                    prof["model_name"] = m_model
                    if "selected_fields" in prof:
                        prof["selected_fields"] = [self.get_val("mining_fields", f, f) for f in prof["selected_fields"]]
                elif prof.get("model_name") in ["Grammar", g_model]:
                    prof["model_name"] = g_model
                    if "selected_fields" in prof:
                        prof["selected_fields"] = [self.get_val("grammar_fields", f, f) for f in prof["selected_fields"]]

            # Update injection profiles
            for prof_name, prof in data.get("profiles", {}).get("injection", {}).items():
                if prof.get("model_name") in ["miningsimple", m_model]:
                    prof["model_name"] = m_model
                    if "target_fields" in prof:
                        prof["target_fields"] = [self.get_val("mining_fields", f, f) for f in prof["target_fields"]]
                    if "field_mapping" in prof:
                        new_mapping = {}
                        for k, v in prof["field_mapping"].items():
                            nk = self.get_val("mining_fields", k, k)
                            nv = self.get_val("mining_fields", v, v)
                            new_mapping[nk] = nv
                        prof["field_mapping"] = new_mapping
                elif prof.get("model_name") in ["Grammar", g_model]:
                    prof["model_name"] = g_model
                    if "target_fields" in prof:
                        prof["target_fields"] = [self.get_val("grammar_fields", f, f) for f in prof["target_fields"]]
                    if "field_mapping" in prof:
                        new_mapping = {}
                        for k, v in prof["field_mapping"].items():
                            nk = self.get_val("grammar_fields", k, k)
                            nv = self.get_val("grammar_fields", v, v)
                            new_mapping[nk] = nv
                        prof["field_mapping"] = new_mapping

            return json.dumps(data, indent=4, ensure_ascii=False)

        self.replace_in_file(fpath, transform)

    def refactor_browser_extensions(self):
        """Refactors browseraddons (grammaraddonregex, markierer_extension, etc.)."""
        # 1. prepare_data.py
        prep_py = os.path.join(self.repo_dir, "browseraddons", "grammaraddonregex", "prepare_data.py")
        lvl_pt = self.get_val("grammar_fields", "Level And Grammar Point")
        link_f = self.get_val("grammar_fields", "Link")
        const_f = self.get_val("grammar_fields", "construction")
        ex_f = self.get_val("grammar_fields", "examplesentences")
        reg_f = self.get_val("grammar_fields", "regexpattern")

        def transform_prep(text):
            text = re.sub(r"row\.get\('Level And Grammar Point',\s*''\)", f"row.get('{lvl_pt}', '')", text)
            text = re.sub(r"row\.get\('Link',\s*''\)", f"row.get('{link_f}', '')", text)
            text = re.sub(r"row\.get\('construction',\s*''\)", f"row.get('{const_f}', '')", text)
            text = re.sub(r"row\.get\('examplesentences',\s*''\)", f"row.get('{ex_f}', '')", text)
            text = re.sub(r"row\.get\('regexpattern',\s*''\)", f"row.get('{reg_f}', '')", text)
            return text

        self.replace_in_file(prep_py, transform_prep)

        # 2. content.js
        content_js = os.path.join(self.repo_dir, "browseraddons", "grammaraddonregex", "content.js")
        nid_f = self.get_val("mining_fields", "Note ID")
        word_f = self.get_val("mining_fields", "Word")
        sp_f = self.get_val("mining_fields", "SentencePlain")
        edo_f = self.get_val("mining_fields", "English Definition Overview")
        freq_f = self.get_val("mining_fields", "Frequency")
        cjd_f = self.get_val("mining_fields", "Correct Japanese Definition")

        def transform_content_js(text):
            # Table headers
            th_pattern = r'<th>Note ID</th>\s*<th>Word</th>\s*<th>SentencePlain</th>\s*<th>English Definition Overview</th>\s*<th>Frequency</th>\s*<th>Correct Japanese Definition</th>'
            th_replace = f'<th>{nid_f}</th>\n        <th>{word_f}</th>\n        <th>{sp_f}</th>\n        <th>{edo_f}</th>\n        <th>{freq_f}</th>\n        <th>{cjd_f}</th>'
            text = re.sub(th_pattern, th_replace, text)

            # TSV Export Header Array
            tsv_header_pattern = r'\["Note ID",\s*"Word",\s*"SentencePlain",\s*"English Definition Overview",\s*"Frequency",\s*"Correct Japanese Definition"\]'
            tsv_header_replace = f'["{nid_f}", "{word_f}", "{sp_f}", "{edo_f}", "{freq_f}", "{cjd_f}"]'
            text = re.sub(tsv_header_pattern, tsv_header_replace, text)

            # Stats Table Header
            text = re.sub(r'<th>Grammar \("Level And Grammar Point"\)</th>', f'<th>Grammar ("{lvl_pt}")</th>', text)
            return text

        self.replace_in_file(content_js, transform_content_js)

        # 3. options.js
        options_js = os.path.join(self.repo_dir, "browseraddons", "grammaraddonregex", "options.js")
        notes_f = self.get_val("grammar_fields", "Notes")

        def transform_options_js(text):
            text = re.sub(r"item\['Level And Grammar Point'\]", f"item['{lvl_pt}']", text)
            text = re.sub(r"g\['Notes'\]", f"g['{notes_f}']", text)

def expand_regex_options(pattern_str):
    """
    Resolves complex, deeply nested regex options like (?:A|B) or (A|B)
    step-by-step from inside to outside into individual plain-text combinations.
    """
    if not pattern_str or pd.isna(pattern_str):
        return []
        
    pattern_str = str(pattern_str).strip()
    
    def replace_char_class(match):
        chars = match.group(1)
        # Prevents splitting predefined regex classes like \s
        if chars.startswith('\\'):
            return match.group(0)
        return '(?:' + '|'.join(list(chars)) + ')'
    
    if '[' in pattern_str and '^' not in pattern_str:
        pattern_str = re.sub(r'\[([^\]]+)\]', replace_char_class, pattern_str)

    def expand_step(text):
        match = re.search(r'\(\?(?::)?([^()]*)\)(\?)?|\(([^()]*)\)(\?)?', text)
        if not match:
            if '|' in text:
                return [opt.strip() for opt in text.split('|') if opt.strip()]
            return [text]
        
        start, end = match.span()
        if match.group(1) is not None:
            inner_content = match.group(1)
            is_optional = match.group(2) == '?'
        else:
            inner_content = match.group(3)
            is_optional = match.group(4) == '?'
            
        alternatives = [opt.strip() for opt in inner_content.split('|') if opt.strip()]
        if is_optional:
            alternatives.append("")
            
        results = []
        prefix = text[:start]
        suffix = text[end:]
        
        for alt in alternatives:
            combined = prefix + alt + suffix
            results.extend(expand_step(combined))
            
        return results

    expanded_patterns = expand_step(pattern_str)
    
    unique_patterns = []
    for p in expanded_patterns:
        p_cleaned = p.strip()
        if p_cleaned and p_cleaned not in unique_patterns:
            unique_patterns.append(p_cleaned)
            
    return unique_patterns

def calculate_max_length_score(pattern_str):
    """
    Calculates the maximum potential match length of a flat pattern.
    """
    score = 0
    temp_str = pattern_str
    quantifiers = re.findall(r'\{\d+,(\d+)\}', temp_str)
    for q in quantifiers:
        score += int(q)
    temp_str = re.sub(r'\{\d+,\d+\}', '', temp_str)
    temp_str = re.sub(r'\[[^\]]+\]', '', temp_str)
    temp_str = re.sub(r'[\(\)\?\+\*^$]', '', temp_str)
    score += len(temp_str)
    return score

def main():
    # Ensure the data directory exists
    if not os.path.exists(DATA_DIR):
        os.makedirs(DATA_DIR)

    if not os.path.exists(TSV_FILE):
        print(f"Error: {TSV_FILE} was not found in the current directory!")
        return

    print("Reading original TSV file...")
    df = pd.read_csv(TSV_FILE, sep='\t', encoding='utf-8-sig').fillna('')
    
    grammar_dict = {}
    pattern_to_nids = {}
    
    for _, row in df.iterrows():
        nid = str(row['nid']).strip()
        if not nid or nid == 'nan':
            continue
            
        # Keep original entry in grammar_dict under its original NID
        grammar_dict[nid] = {
            "nid": nid,
            "tags": row.get('tags', ''),
            "level_and_point": row.get('Level And Grammar Point', ''),
            "link": row.get('Link', ''),
            "construction": row.get('construction', ''),
            "examplesentences": row.get('examplesentences', ''),
            "regexpattern": row.get('regexpattern', '')
        }
        
        raw_regex = row.get('regexpattern', '')
        flat_patterns = expand_regex_options(raw_regex)
        tags_str = str(row.get('tags', ''))
        
        for pat in flat_patterns:
            if pat not in pattern_to_nids:
                pattern_to_nids[pat] = set()
            
            # Always add the original NID

            elif "そうexpressions" in tags_str:

                pattern_to_nids[pat].update([
                    "1775895481203",

                    "1785181057952"
                ])
            
    processed_patterns = []
    for pat, nids in pattern_to_nids.items():

        processed_patterns.append({
            "pattern": pat,

import os

def verarbeite_tsv(tsv_pfad, ziel_pfad):
    # 1. Read words from the TSV file
    woerter = []
    with open(tsv_pfad, 'r', encoding='utf-8') as f:
        for zeile in f:
            # Removes line breaks and whitespace at the beginning/end of the line
            wort = zeile.strip()
            # Only add if the line is not empty (e.g., the header line "Word" or blank lines)
            if wort:
                woerter.append(wort)
    
    if not woerter:
        print("No words found in the TSV file.")
        return

    # 2. Create string: Each word gets a "@" prefix
    # Example: ['A', 'B'] becomes "@A@B"
    ergebnis_string = "".join(f"@{wort}" for wort in woerter)
    
    # 3. Append seamlessly to the target file ('a' stands for append)
    # Since we use 'write()' and do not add a line break,
    # the string is appended seamlessly to the end of the file.
    with open(ziel_pfad, 'a', encoding='utf-8') as f:
        f.write(ergebnis_string)

    print(f"Successfully appended {len(woerter)} words seamlessly to '{ziel_pfad}'.")

    text = re.sub(r'「.*?」', '', text)
    text = re.sub(r'^[A-Za-z0-9_À-ÿ\s]+:\s*', '', text)
    text = re.sub(r'[\s\t\n\r\.,\?\!。、？！…♪～\-—「」\(\)（）"’]+', '', text)
    return text.strip().lower()

def generiere_lueckenlose_teilstrings(text):
    laenge = len(text)
    teilstrings = []
    for fenster_groesse in range(laenge, 1, -1):
        for start_pos in range(laenge - fenster_groesse + 1):
            unterstring = text[start_pos : start_pos + fenster_groesse]
            if unterstring not in teilstrings:
                teilstrings.append(unterstring)
    return teilstrings

def print_progress_bar(iteration, total, prefix='', suffix='', decimals=1, length=40, fill='█'):
    percent = ("{0:." + str(decimals) + "f}").format(100 * (iteration / float(total)))
    filled_length = int(length * iteration // total)
    bar = fill * filled_length + '░' * (length - filled_length)
    sys_write = f'\r{prefix} [{bar}] {percent}% {suffix}'
    print(sys_write, end='\r')
    if iteration == total:
        print()

def clean_text_subs(text):
    if not text:
        return ""
    text = re.sub(r"\{[^}]+\}", "", text)
    text = text.replace(r"\N", "").replace(r"\n", "")
    text = re.sub(r"[\s●③②①④⑤⑥⑦⑧⑨⓪①-⑨\.。,、 („)“、「」『』？?！!！？\-=_+*⑨<>█░…·•\"\'’()（）]", "", text)
    return text

def get_clean_map(original_text):
    clean_chars = []
    mapping = []
    i = 0
    n = len(original_text)
    while i < n:
        if original_text[i] == '{':
            end = original_text.find('}', i)
            if end != -1:
                i = end + 1
                continue
        if i + 1 < n and original_text[i] == '\\' and original_text[i+1].lower() == 'n':
            i += 2
            continue
        
        clean_chars.append(original_text[i])
        mapping.append(i)
        i += 1
        
    return "".join(clean_chars), mapping

def window_search_match(tsv_text, ass_text, min_ratio=0.55):
    if not tsv_text or not ass_text:
        return False
    if len(ass_text) <= len(tsv_text):
        return SequenceMatcher(None, tsv_text, ass_text).ratio() >= MATCH_THRESHOLD

    window_size = len(tsv_text)
    best_ratio = 0.0
    for i in range(len(ass_text) - window_size + 1):
        sub_str = ass_text[i:i+window_size]
        ratio = SequenceMatcher(None, tsv_text, sub_str).ratio()
        if ratio > best_ratio:
            best_ratio = ratio
            if best_ratio >= 0.80:
                return True
    return best_ratio >= min_ratio

def find_timestamps_and_text_gleitend(srt_subs, target_sentence, start_index=0, delay_ms=0):
    cleaned_target = clean_text_temporary(target_sentence)
    if not cleaned_target:
        return None, None, start_index

    such_begriffe = generiere_lueckenlose_teilstrings(cleaned_target)
    total_subs = len(srt_subs)

    for begriff in such_begriffe:
        for i in range(start_index, total_subs):
            sub_text_cleaned = clean_text_temporary(srt_subs[i].text)
            
            if begriff in sub_text_cleaned:
                start_block_idx = i
                end_block_idx = i
                current_combined_cleaned = sub_text_cleaned
                
                # 1. BACKWARD SCAN
                for prev_j in range(i - 1, max(-1, start_index - 1, i - 11), -1):
                    prev_cleaned = clean_text_temporary(srt_subs[prev_j].text)
                    if not prev_cleaned:
                        continue
                    potential_match = prev_cleaned + current_combined_cleaned
                    if prev_cleaned in cleaned_target and potential_match in cleaned_target:
                        start_block_idx = prev_j
                        current_combined_cleaned = potential_match
                    else:
                        break
                
                # 2. FORWARD SCAN
                for next_j in range(i + 1, min(total_subs, i + 11)):
                    next_cleaned = clean_text_temporary(srt_subs[next_j].text)
                    if not next_cleaned:
                        continue
                    potential_match = current_combined_cleaned + next_cleaned
                    if next_cleaned in cleaned_target and (potential_match in cleaned_target or cleaned_target.endswith(next_cleaned)):
                        end_block_idx = next_j
                        current_combined_cleaned = potential_match
                    else:
                        break
                
                start_time = srt_subs[start_block_idx].start
                end_time = srt_subs[end_block_idx].end
                
                start_ms = (start_time.hours * 3600 + start_time.minutes * 60 + start_time.seconds) * 1000 + start_time.milliseconds
                end_ms = (end_time.hours * 3600 + end_time.minutes * 60 + end_time.seconds) * 1000 + end_time.milliseconds
                
                start_ms += delay_ms
                end_ms += delay_ms
                
                if start_ms < 0: start_ms = 0
                if end_ms < 0: end_ms = 0
                
                return start_ms, end_ms, end_block_idx + 1

    if start_index > 0:
        return find_timestamps_and_text_gleitend(srt_subs, target_sentence, start_index=0, delay_ms=delay_ms)
                
    return None, None, start_index

def convert_time_to_ass(seconds):
    hours = int(seconds // 3600)
    minutes = int((seconds % 3600) // 60)
    secs = int(seconds % 60)
    centiseconds = int(round((seconds % 1) * 100))
    if centiseconds == 100:
        centiseconds = 99
    return f"{hours}:{minutes:02d}:{secs:02d}.{centiseconds:02d}"

def wrap_text_ass(text, max_chars=40):
    if not text:
        return ""
    lines = []
    for line in text.splitlines():
        line = line.strip()
        if not line:
            continue
        current = []
        count = 0
        for char in line:
            current.append(char)
            count += 1
            if count >= max_chars:
                lines.append("".join(current))
                current = []
                count = 0
        if current:
            lines.append("".join(current))
    return r"\N".join(lines)

def convert_srt_to_ass(srt_path, output_ass_path):
    print(f"Creating 3-Block ASS Subtitle: {os.path.basename(srt_path)}...")
    subs = pysrt.open(srt_path)
    
    raw_blocks = []
    for sub in subs:
        start_ms = (sub.start.hours * 3600 + sub.start.minutes * 60 + sub.start.seconds) * 1000 + sub.start.milliseconds
        end_ms = (sub.end.hours * 3600 + sub.end.minutes * 60 + sub.end.seconds) * 1000 + sub.end.milliseconds
        raw_blocks.append({
            "start": start_ms,
            "end": end_ms,
            "text": sub.text.replace('\n', ' ')
        })

    blocks = []
    for i in range(0, len(raw_blocks), 3):
        group = raw_blocks[i:i+3]
        combined_text = " ".join([b["text"] for b in group])
        combined_start = group[0]["start"]
        combined_end = group[-1]["end"]
        blocks.append({
            "start": combined_start,
            "end": combined_end,
            "text": combined_text
        })

    ass_content = [
        "[Script Info]",
        "Title: 3-Block Left Aligned",
        "ScriptType: v4.00+",
        "PlayResX: 1920",
        "PlayResY: 1080",
        "WrapStyle: 0",
        "ScaledBorderAndShadow: yes",
        "",
        "[V4+ Styles]",
        "Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding",
        f"Style: Text,MS Gothic,38,&H00000000,&H000000FF,&H00FFFFFF,&H00000000,0,0,0,0,100,100,0,0,1,8,0,7,20,20,20,1",
        f"Style: Timer,MS Gothic,42,&H00000000,&H000000FF,&H00FFFFFF,&H00000000,0,0,0,0,100,100,0,0,1,8,0,7,20,20,20,1",
        "",
        "[Events]",
        "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"
    ]

    pos_tag = r"{\pos"

    for i in range(len(blocks)):
        current = blocks[i]
        start_ass = convert_time_to_ass(current["start"] / 1000)
        end_ass = convert_time_to_ass(current["end"] / 1000)
        duration = int((current["end"] - current["start"]) / 1000)

        if i - 1 >= 0:
            ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{pos_tag}(120,{Y_PREV})}}{wrap_text_ass(blocks[i-1]["text"])}')
        
        current_text = wrap_text_ass(current["text"])
        ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{pos_tag}(120,{Y_ACTIVE})}}●{current_text}')
        
        if i + 1 < len(blocks):
            ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{pos_tag}(120,{Y_NEXT})}}{wrap_text_ass(blocks[i+1]["text"])}')

        for sec in range(duration):
            t_start = convert_time_to_ass((current["start"] / 1000) + sec)
            t_end = convert_time_to_ass((current["start"] / 1000) + sec + 1)
            remaining = duration - sec
            ass_content.append(f'Dialogue: 5,{t_start},{t_end},Timer,,0,0,0,,{{\\pos(1750,{Y_TIMER})}}{remaining}')

    with open(output_ass_path, "w", encoding="utf-8") as f:
        f.write("\n".join(ass_content))

def extract_video_frame(video_path, timestamp_ms, output_path):
    if not OPENCV_AVAILABLE:
        return False
    try:
        cap = cv2.VideoCapture(video_path)
        if not cap.isOpened():
            return False
        
        cap.set(cv2.CAP_PROP_POS_MSEC, timestamp_ms)
        success, frame = cap.read()
        if success:
            cv2.imwrite(output_path, frame, [int(cv2.IMWRITE_JPEG_QUALITY), 90])
            cap.release()
            return True
        cap.release()
    except Exception as e:
        print(f"Error capturing screenshot: {e}")
    return False

def process_single_triple(table_path, sub_path, media_path, mode, quality, want_screenshots, output_dir, backup_dir):
    """Core logic to process a single triple of (TSV, SRT, Media)"""
    run_media = (mode in ["1", "2"])
    run_subs = (mode in ["1", "3"])

    base_name = os.path.splitext(os.path.basename(table_path))[0]
    orig_dir = os.path.dirname(table_path)

    subs_pysrt = pysrt.open(sub_path) if sub_path and os.path.exists(sub_path) else None

    # Read table file
    with open(table_path, mode="r", encoding="utf-8-sig") as f:
        sample = f.read(2048)
        delimiter = "\t" if "\t" in sample else (";" if ";" in sample else ",")

    table_rows = []
    with open(table_path, mode="r", encoding="utf-8-sig") as f:
        reader = csv.DictReader(f, delimiter=delimiter)
        fieldnames = list(reader.fieldnames) if reader.fieldnames else []
        for row in reader:
            table_rows.append({k.strip(): v.strip() if v else "" for k, v in row.items() if k})

    # --- PART 1: AUDIO & SCREENSHOT EXTRACTION ---
    if run_media and media_path and subs_pysrt:
        if AUDIO_FRONT_COLUMN not in fieldnames: fieldnames.append(AUDIO_FRONT_COLUMN)
        if AUDIO_BACK_COLUMN not in fieldnames: fieldnames.append(AUDIO_BACK_COLUMN)
        if want_screenshots and IMAGE_COLUMN_NAME not in fieldnames: fieldnames.append(IMAGE_COLUMN_NAME)

        print(f"\n[PART 1] Starting media processing for {base_name}...")
        full_audio = AudioSegment.from_file(media_path)
        audio_duration_ms = len(full_audio)

        extracted_map = {}
        last_srt_index = 0

        for row in table_rows:
            raw_sentence = row.get(SENTENCE_COLUMN_NAME, "")
            cleaned_sentence = clean_text_temporary(raw_sentence)

            if not cleaned_sentence: continue

            if cleaned_sentence in extracted_map:
                r_name = extracted_map[cleaned_sentence]
                sound_tag = f"[sound:{r_name}.mp3]"
                row[AUDIO_FRONT_COLUMN] = sound_tag
                row[AUDIO_BACK_COLUMN] = sound_tag
                if want_screenshots: row[IMAGE_COLUMN_NAME] = f'<img src="{r_name}.jpg">'
                continue

            # Sliding substring search matching
            start_ms, end_ms, next_index = find_timestamps_and_text_gleitend(subs_pysrt, raw_sentence, last_srt_index)

            if start_ms is not None:
                last_srt_index = next_index
                r_name = generate_random_name()
                extracted_map[cleaned_sentence] = r_name

                # PADDING & MINIMUM DURATION LOGIC (MIN. 15 SECONDS EQUALLY PADDED)
                audio_start_ms = max(0, start_ms - 2000)
                audio_end_ms = min(audio_duration_ms, end_ms + 2000)

                current_duration = audio_end_ms - audio_start_ms
                MIN_DURATION_MS = 15000  # 15 Seconds

                if current_duration < MIN_DURATION_MS:
                    needed_extra = MIN_DURATION_MS - current_duration
                    pad_left = needed_extra // 2
                    pad_right = needed_extra - pad_left
                    
                    audio_start_ms -= pad_left
                    audio_end_ms += pad_right

                    if audio_start_ms < 0:
                        audio_end_ms = min(audio_duration_ms, audio_end_ms + abs(audio_start_ms))
                        audio_start_ms = 0

                    if audio_end_ms > audio_duration_ms:
                        audio_start_ms = max(0, audio_start_ms - (audio_end_ms - audio_duration_ms))
                        audio_end_ms = audio_duration_ms

                output_audio_path = os.path.join(output_dir, f"{r_name}.mp3")
                
                # Export audio clip
                audio_segment = full_audio[audio_start_ms:audio_end_ms]
                audio_segment = audio_segment.set_channels(quality["channels"]).set_frame_rate(quality["frame_rate"])
                audio_segment.export(output_audio_path, format=quality["format"], bitrate=quality["bitrate"])

                sound_tag = f"[sound:{r_name}.mp3]"
                row[AUDIO_FRONT_COLUMN] = sound_tag
                row[AUDIO_BACK_COLUMN] = sound_tag

                # Export screenshot (without padding)
                if want_screenshots:
                    output_image_path = os.path.join(output_dir, f"{r_name}.jpg")
                    if extract_video_frame(media_path, start_ms, output_image_path):
                        row[IMAGE_COLUMN_NAME] = f'<img src="{r_name}.jpg">'

        print(f"Media extraction finished for {base_name}.")

    # --- SAVE UPDATED OUTPUT TSV WITH SUFFIX _output ---
    output_tsv_path = os.path.join(orig_dir, f"{base_name}_output.tsv")
    with open(output_tsv_path, mode="w", encoding="utf-8-sig", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames, delimiter=delimiter)
        writer.writeheader()
        writer.writerows(table_rows)
    print(f"New output TSV created: {output_tsv_path}")

    # --- PART 2: 3-BLOCK ASS SUBTITLE CREATION & HIGHLIGHTING ---
    if run_subs and sub_path:
        print(f"\n[PART 2] Generating and Formatting ASS subtitles for {base_name}...")
        ass_path = os.path.splitext(sub_path)[0] + ".ass"
        
        # 1. Base Conversion
        convert_srt_to_ass(sub_path, ass_path)

        for row in table_rows:
            word = row.get(WORD_COLUMN_NAME, "")
            sentence = row.get(SENTENCE_COLUMN_NAME, "")

                continue

            bullet_idx = text_part.find("●")
            if bullet_idx != -1:
                prefix_part = text_part[:bullet_idx + 1]

                            'start': idx,
                            'end': idx + len(word),
                            'word': word,
                            'color': color,
                            'length': len(word)
                        })
                        start_pos = idx + 1

                style_at_clean = [None] * len(clean_text_for_locating)
                for j in range(len(clean_text_for_locating)):
                    covering = [inv for inv in intervals if inv['start'] <= j < inv['end']]
                    if covering:
                        covering.sort(key=lambda x: x['length']) 
                        style_at_clean[j] = covering[0]

                orig_style = {}
                for j, orig_idx in enumerate(mapping):
                    orig_style[orig_idx] = style_at_clean[j]

                reconstructed = []
                current_style = None
                k = 0
                n_orig = len(text_to_process)

                while k < n_orig:

def scan_batch_triples(selected_dir):
    """Scans the directory tree for matching TSV, SRT, and Video/Media triples"""
    valid_triples = []
    orphans = []
    
    media_extensions = ('.mp4', '.mkv', '.m4a', '.mp3', '.wav', '.avi', '.webm')

    for root, dirs, files in os.walk(selected_dir):
        tsv_map = {}
        srt_map = {}
        media_map = {}

        for f in files:
            ext = os.path.splitext(f)[1].lower()
            base = os.path.splitext(f)[0]

            # Ignore previously processed/output files
            if ext in ['.tsv', '.csv']:
                if base.endswith('_processed') or base.endswith('_output') or base.startswith('output_'):
                    continue
                tsv_map[base] = os.path.join(root, f)
            elif ext == '.srt':
                srt_map[base] = os.path.join(root, f)
            elif ext in media_extensions:
                media_map[base] = os.path.join(root, f)

        tsv_set = set(tsv_map.keys())
        srt_set = set(srt_map.keys())
        media_set = set(media_map.keys())

        matched_bases = tsv_set.intersection(srt_set).intersection(media_set)

        for base in matched_bases:
            valid_triples.append({
                'base': base,
                'tsv_path': tsv_map[base],
                'srt_path': srt_map[base],
                'media_path': media_map[base],
                'folder': root
            })

        # Track incomplete files for warnings
        all_found = tsv_set.union(srt_set).union(media_set)
        for base in all_found - matched_bases:
            missing = []
            if base not in tsv_set: missing.append("TSV")
            if base not in srt_set: missing.append("SRT")
            if base not in media_set: missing.append("Media")
            orphans.append(f"Incomplete matching for '{base}' in '{root}' (Missing: {', '.join(missing)})")

    return valid_triples, orphans

def convert_time_to_ass(seconds):
    hours = int(seconds // 3600)
    minutes = int((seconds % 3600) // 60)
    secs = int(seconds % 60)
    centiseconds = int(round((seconds % 1) * 100))
    if centiseconds == 100:
        centiseconds = 99
    return f"{hours}:{minutes:02d}:{secs:02d}.{centiseconds:02d}"


def wrap_text(text, max_chars=45):
    """Hard line wrap per original line"""
    if not text:
        return ""
    
    lines = []
    
    # Process line by line (respecting existing line breaks)
    for line in text.splitlines():
        line = line.strip()
        if not line:
            continue
            
        current = []
        count = 0
        
        for char in line:
            current.append(char)
            count += 1
            
            if count >= max_chars:
                lines.append("".join(current))
                current = []
                count = 0
                
        if current:
            lines.append("".join(current))
    
    return r"\N".join(lines)


def txt_to_ass(input_path):

            if not line:
                continue
            match = pattern.match(line)
            if match:
                start, end, text = match.groups()
                blocks.append({
                    "start": float(start),
                    "end": float(end),
                    "text": text.strip()
                })

    ass_content = [
        "[Script Info]",

        "Style: Timer,MS Gothic,42,&H00FFFF00,&H000000FF,&H00000000,&H80000000,0,0,0,0,100,100,0,0,1,2,0,7,20,20,20,1",
        "",
        "[Events]",
        "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"
    ]

    for i in range(len(blocks)):
        current = blocks[i]
        start_ass = convert_time_to_ass(current["start"])
        end_ass = convert_time_to_ass(current["end"])
        duration = int(current["end"] - current["start"])

        # Left-aligned positions (Alignment 7)
        if i - 2 >= 0:
            ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{{\\pos(120,100)}}{wrap_text(blocks[i-2]["text"])}')

        if i - 1 >= 0:
            ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{{\\pos(120,300)}}{wrap_text(blocks[i-1]["text"])}')
        
        # Current block
        current_text = wrap_text(current["text"])
        ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{{\\pos(120,500)}}●{current_text}')

            remaining = duration - sec
            ass_content.append(f'Dialogue: 5,{t_start},{t_end},Timer,,0,0,0,,{{\\pos(1750,490)}}{remaining}')

    with open(output_path, "w", encoding="utf-8") as f:
        f.write("\n".join(ass_content))

    print(f"✅ ASS file successfully generated: {output_path}")

def to_hiragana(text):
    if not text: return ""
    trans_table = str.maketrans(
        'ァアィイゥウェエォオカガキギクグケゲコゴサザシジスズセゼソゾタダチヂッツヅテデトドナニヌネノハバパヒビピフブプヘベペホボポマミムメモヤャユュヨョラリルレロワヮヰヱヲンヴヵヶ',
        'ぁあぃいううぇえぉおかがきぎくぐけげこごさざしじすずせぜそぞただちぢっつづてでとどなにぬねのはばぱひびぴふぶぷへべぺほぼぽまみむめもやゃゆゅよょらりるれろわゎゐゑをんゔゕゖ'
    )
    text = text.translate(trans_table)
    result = kks.convert(text)
    hira = "".join([item['hira'] for item in result])
    hira = hira.translate(trans_table)
    hira = re.sub(r'[^\u3040-\u309Fー]', '', hira)
    return hira


def find_best_match_in_area(target, ki_full_string, search_start, search_end):
    search_area = ki_full_string[search_start:search_end]
    if not target or not search_area:
        return 0, search_start, ""
    
    best_score = 0
    best_ki_idx = search_start
    window_size = min(40, len(target))
    
    for i in range(0, len(search_area) - window_size + 1, 1):
        candidate = search_area[i:i + window_size]
        score = SequenceMatcher(None, target[:window_size], candidate).ratio()
        if score > best_score:
            value_modifier = 1.0
            if i + window_size >= len(search_area) - 5:
                value_modifier = 1.15
            
            modified_score = score * value_modifier
            if modified_score > best_score:
                best_score = min(1.0, modified_score)
                best_ki_idx = search_start + i

    if best_score < MATCH_THRESHOLD:
        for n in range(12, 5, -1):
            if len(target) < n: continue
            ngram = target[:n]
            if ngram in search_area:
                pos = search_area.index(ngram)
                candidate = search_area[pos:pos + len(target)]
                score = SequenceMatcher(None, target, candidate).ratio()
                if score > best_score:
                    best_score = score
                    best_ki_idx = search_start + pos
                    break
                    
    return best_score, best_ki_idx, ki_full_string[best_ki_idx:best_ki_idx + len(target)]


def process_from_json(json_path, text_path, output_folder, base_name):
    output_path = os.path.join(output_folder, f"{base_name}_zeitmarken.txt")
    log_path = os.path.join(output_folder, f"{base_name}_mapping_log.txt")
    json_log_path = os.path.join(output_folder, f"{base_name}_mapping_details.json")

    print(f"\n--- Processing from JSON source: {base_name} ---")

    # 1. Load text and consolidate blocks
    print("⏳ Loading and processing script text file...")
    with open(text_path, "r", encoding="utf-8") as f:
        raw_lines = [line.strip() for line in f if line.strip()]

    consolidated_blocks, block_hira_list = [], []
    current_block = ""
    for line in raw_lines:
        current_block = (current_block + " " + line).strip()
        if len(current_block) >= MIN_CHAR_COUNT:
            consolidated_blocks.append(current_block)
            block_hira_list.append(to_hiragana(current_block))
            current_block = ""
    if current_block:
        consolidated_blocks.append(current_block)
        block_hira_list.append(to_hiragana(current_block))

    # 2. Load existing AI transcription from JSON
    print(f"⏳ Loading AI segments from: {os.path.basename(json_path)}")
    with open(json_path, "r", encoding="utf-8") as f:
        ai_segments = json.load(f)

    # 3. Create AI timeline (Granular down to character level)
    print("⏳ Generating AI timeline on character level...")
    ki_timeline = []
    for seg in ai_segments:
        hira = to_hiragana(seg['text'])
        if not hira: continue
        if len(hira) > 5 and len(set(hira)) <= 2: continue

        seg_duration = seg['end'] - seg['start']
        step = seg_duration / len(hira) if len(hira) > 0 else 0
        
        consecutive_count = 0
        for i, char in enumerate(hira):
            consecutive_count = (consecutive_count + 1) if ki_timeline and char == ki_timeline[-1]["char"] else 1
            if consecutive_count > 3: continue
            ki_timeline.append({
                "char": char, 
                "t": seg['start'] + (i * step),
                "seg_end": seg['end'],
                "seg_text": hira
            })

    ki_full_string = "".join([x["char"] for x in ki_timeline])
    total_ki_len = len(ki_timeline)

    if total_ki_len == 0:
        print("❌ ERROR: No valid AI data found in JSON. Aborting.")
        return

    # 4. Timestamp mapping with two-way search
    print(f"\n🚀 Starting timestamp mapping ({len(consolidated_blocks)} blocks to map)...")
    final_output = []
    mapping_logs = []
    json_mapping_details = []
    
    last_ki_index = 0
    last_end_time = 0.0
    block_idx = 0
    num_blocks = len(consolidated_blocks)

    while block_idx < num_blocks:
        block = consolidated_blocks[block_idx]
        block_hira = block_hira_list[block_idx]
        block_len = len(block_hira)

        search_start = max(0, last_ki_index - 100)
        search_end = min(total_ki_len, last_ki_index + 1200)

        # Progress log every 10 blocks or at difficult spots
        if block_idx % 10 == 0 or block_idx == num_blocks - 1:
            prozent = (block_idx / num_blocks) * 100

def convert_time_to_ass(seconds):
    hours = int(seconds // 3600)
    minutes = int((seconds % 3600) // 60)
    secs = int(seconds % 60)
    centiseconds = int(round((seconds % 1) * 100))
    if centiseconds == 100:
        centiseconds = 99
    return f"{hours}:{minutes:02d}:{secs:02d}.{centiseconds:02d}"


def wrap_text(text, max_chars=45):
    """Hard line wrap per original line"""
    if not text:
        return ""
    
    lines = []
    
    # Process line by line (respects existing line breaks)
    for line in text.splitlines():
        line = line.strip()
        if not line:
            continue
            
        current = []
        count = 0
        
        for char in line:
            current.append(char)
            count += 1
            
            if count >= max_chars:
                lines.append("".join(current))
                current = []
                count = 0
                
        if current:
            lines.append("".join(current))
    
    return r"\N".join(lines)


def txt_to_ass(input_path):
    base_path, _ = os.path.splitext(input_path)
    output_path = base_path + ".ass"

    pattern = re.compile(r"\[\s*([\d.]+)\s*->\s*([\d.]+)\s*\]\s*(.*)")
    blocks = []

    with open(input_path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            match = pattern.match(line)
            if match:
                start, end, text = match.groups()
                blocks.append({
                    "start": float(start),
                    "end": float(end),
                    "text": text.strip()
                })

    ass_content = [
        "[Script Info]",
        "Title: 5-Block Left Aligned",
        "ScriptType: v4.00+",
        "PlayResX: 1920",
        "PlayResY: 1080",
        "WrapStyle: 0",
        "ScaledBorderAndShadow: yes",
        "",
        "[V4+ Styles]",
        "Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding",
        "Style: Text,MS Gothic,38,&H00FFFFFF,&H000000FF,&H00000000,&H80000000,0,0,0,0,100,100,0,0,1,2,0,7,20,20,20,1",
        "Style: Timer,MS Gothic,42,&H00FFFF00,&H000000FF,&H00000000,&H80000000,0,0,0,0,100,100,0,0,1,2,0,7,20,20,20,1",
        "",
        "[Events]",
        "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text"
    ]

    for i in range(len(blocks)):
        current = blocks[i]
        start_ass = convert_time_to_ass(current["start"])
        end_ass = convert_time_to_ass(current["end"])
        duration = int(current["end"] - current["start"])

        # Left-aligned positions (Alignment 7)
        if i - 2 >= 0:
            ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{{\\pos(120,100)}}{wrap_text(blocks[i-2]["text"])}')

        if i - 1 >= 0:
            ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{{\\pos(120,300)}}{wrap_text(blocks[i-1]["text"])}')
        
        # Current block
        current_text = wrap_text(current["text"])
        ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{{\\pos(120,500)}}●{current_text}')
        
        # Next block
        if i + 1 < len(blocks):
            ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{{\\pos(120,700)}}{wrap_text(blocks[i+1]["text"])}')

        # Block after next
        if i + 2 < len(blocks):
            ass_content.append(f'Dialogue: 0,{start_ass},{end_ass},Text,,0,0,0,,{{\\pos(120,900)}}{wrap_text(blocks[i+2]["text"])}')

        # Countdown
        for sec in range(duration):
            t_start = convert_time_to_ass(current["start"] + sec)
            t_end = convert_time_to_ass(current["start"] + sec + 1)
            remaining = duration - sec
            ass_content.append(f'Dialogue: 5,{t_start},{t_end},Timer,,0,0,0,,{{\\pos(1750,490)}}{remaining}')

    with open(output_path, "w", encoding="utf-8") as f:
        f.write("\n".join(ass_content))

    print(f"\n✅ Converted (left-aligned + improved text wrapping): {output_path}")

def to_hiragana(text):
    if not text: return ""
    trans_table = str.maketrans(
        'ァアィイゥウェエォオカガキギクグケゲコゴサザシジスズセゼソゾタダチヂッツヅテデトドナニヌネノハバパヒビピフブプヘベペホボポマミムメモヤャユュヨョラリルレロワヮヰヱヲンヴヵヶ',
        'ぁあぃいううぇえぉおかがきぎくぐけげこごさざしじすずせぜそぞただちぢっつづてでとどなにぬねのはばぱひびぴふぶぷへべぺほぼぽまみむめもやゃゆゅよょらりるれろわゎゐゑをんゔゕゖ'
    )
    text = text.translate(trans_table)
    result = kks.convert(text)
    hira = "".join([item['hira'] for item in result])
    hira = hira.translate(trans_table)
    hira = re.sub(r'[^\u3040-\u309Fー]', '', hira)
    return hira


def find_best_match_in_area(target, ki_full_string, search_start, search_end):
    search_area = ki_full_string[search_start:search_end]
    if not target or not search_area:
        return 0, search_start, ""
    
    best_score = 0
    best_ki_idx = search_start
    window_size = min(40, len(target))
    
    for i in range(0, len(search_area) - window_size + 1, 1):
        candidate = search_area[i:i + window_size]
        score = SequenceMatcher(None, target[:window_size], candidate).ratio()
        if score > best_score:
            value_modifier = 1.0
            if i + window_size >= len(search_area) - 5:
                value_modifier = 1.15
            
            modified_score = score * value_modifier
            if modified_score > best_score:
                best_score = min(1.0, modified_score)
                best_ki_idx = search_start + i

    if best_score < MATCH_THRESHOLD:
        for n in range(12, 5, -1):
            if len(target) < n: continue
            ngram = target[:n]
            if ngram in search_area:
                pos = search_area.index(ngram)
                candidate = search_area[pos:pos + len(target)]
                score = SequenceMatcher(None, target, candidate).ratio()
                if score > best_score:
                    best_score = score
                    best_ki_idx = search_start + pos
                    break
                    
    return best_score, best_ki_idx, ki_full_string[best_ki_idx:best_ki_idx + len(target)]

                method_info = "FORCED_LINEAR"
                diagnosis = "Pattern correlates, but index is moving backward. Forced forward movement."
            else:
                method_info = "FUZZY (OK)"

                        mapping_logs.append(f"Block {k:03d} | Method: DELETION_STRETCH\n  Diagnosis: {diagnosis}\n  Time: {s_t:.2f}s to {e_t:.2f}s\n{'-'*50}")
                        running_time = e_t
                    
                    last_end_time = future_match_time
                    last_ki_index = f_matched_idx
                    block_idx = future_idx  
                    found_future_match = True
                    break

            if found_future_match:
                continue

            # --- FALLBACK: IF NOTHING MATCHES IN THE FUTURE (LOOP / HALLUCINATION) ---
            method_info = "LOOP_RECOVERY_FALLBACK"
            estimated_duration = max(2.0, block_len * 0.14)
            target_time = last_end_time + estimated_duration
            diagnosis = "No match found in future book text. Using linear time estimation based on character length."
            
            current_ki_end_idx = last_ki_index
            for idx in range(last_ki_index, total_ki_len):
                if ki_timeline[idx]["t"] >= target_time:
                    current_ki_end_idx = idx
                    break
            if current_ki_end_idx == last_ki_index:
                current_ki_end_idx = min(total_ki_len - 1, last_ki_index + max(5, block_len))

            s_idx = max(0, min(last_ki_index, total_ki_len - 1))
            e_idx = max(0, min(int(current_ki_end_idx), total_ki_len - 1))
            start_t = ki_timeline[s_idx]["t"]
            end_t = ki_timeline[e_idx]["t"]

            if start_t < last_end_time: start_t = last_end_time
            if end_t <= start_t: end_t = start_t + max(1.5, block_len * 0.14)

            final_output.append(f"[{start_t:6.2f} -> {end_t:6.2f}] {block}")
            
            json_mapping_details.append({
                "block_index": block_idx,
                "text_preview": block[:30] + "...",
                "methode": method_info,
                "score": round(score, 4),
                "zeit_start": round(start_t, 2),
                "zeit_end": round(end_t, 2),
                "vergleich": {"original_skript_anker_hiragana": block_hira[:50], "ki_transkript_match_hiragana": ""}
            })
            mapping_logs.append(f"Block {block_idx:03d} | Method: {method_info}\n  Diagnosis: {diagnosis}\n  Time: {start_t:.2f}s to {end_t:.2f}s\n{'-'*50}")
            
            last_ki_index = e_idx
            last_end_time = end_t
            block_idx += 1

    # 5. Write files
    with open(output_path, "w", encoding="utf-8") as f:
        f.writelines(line + "\n" for line in final_output)

    with open(log_path, "w", encoding="utf-8") as f:

    for root, dirs, files in os.walk(selected_dir):
        # Ignore old "_zeitmarken" files in the scan
        txt_files = {
            os.path.splitext(f)[0] for f in files 
            if f.endswith(".txt") and not f.endswith("_zeitmarken.txt")
        }
        
        # NOW: Detects .m4a, .mp3 and .wav (Mapping via the base name)
        audio_map = {
            os.path.splitext(f)[0]: f for f in files 
            if f.lower().endswith(audio_extensions)
        }

        audio_files_set = set(audio_map.keys())

        # Find matching pairs of text and audio files
        for base in txt_files.intersection(audio_files_set):
            valid_pairs.append({
                "base": base,
                "txt_path": os.path.join(root, base + ".txt"),
                "audio_path": os.path.join(root, audio_map[base]), # Uses the exact audio file (whether .mp3 or .m4a)

            })

        for base in txt_files - audio_files_set:
            orphans.append(f"Only text found (No audio): {os.path.join(root, base + '.txt')}")
        for base in audio_files_set - txt_files:
            orphans.append(f"Only audio found (No text): {os.path.join(root, audio_map[base])}")

    if orphans:
        print("\n=== WARNING: Incomplete pairs found ===")
        for orphan in orphans:
            print(orphan)
        print("=======================================\n")

    if not valid_pairs:
        print("No matching pairs found. Ending batch mode.")
        return

    valid_pairs.sort(key=lambda x: x["txt_path"])

    print(f"{len(valid_pairs)} valid pairs were found:\n")
    for idx, pair in enumerate(valid_pairs):

    try:

        print("Invalid input. Please enter a number.")
        return

        print("Index out of valid range.")
        return

    for i in range(start_idx, end_idx + 1):

def get_clean_map(original_text):
    clean_chars = []
    mapping = []
    i = 0
    n = len(original_text)
    while i < n:
        if original_text[i] == '{':
            end = original_text.find('}', i)
            if end != -1:
                i = end + 1
                continue
        if i + 1 < n and original_text[i] == '\\' and original_text[i+1].lower() == 'n':
            i += 2
            continue
        
        clean_chars.append(original_text[i])
        mapping.append(i)
        i += 1
        
    return "".join(clean_chars), mapping

def ass_time_to_ms(time_str):
    try:
        parts = time_str.strip().split(':')
        hours = int(parts[0])
        minutes = int(parts[1])
        seconds_parts = parts[2].split('.')
        seconds = int(seconds_parts[0])
        frac_str = seconds_parts[1]
        if len(frac_str) == 2:
            ms = int(frac_str) * 10
        elif len(frac_str) == 3:
            ms = int(frac_str)
        else:
            ms = int(frac_str[:3].ljust(3, '0'))
        return (hours * 3600 + minutes * 60 + seconds) * 1000 + ms
    except Exception:
        return None

def window_search_match(tsv_text, ass_text, min_ratio=0.55):
    if not tsv_text or not ass_text:
        return False
    if len(ass_text) <= len(tsv_text):
        return SequenceMatcher(None, tsv_text, ass_text).ratio() >= MATCH_THRESHOLD

    window_size = len(tsv_text)
    best_ratio = 0.0
    for i in range(len(ass_text) - window_size + 1):
        sub_str = ass_text[i:i+window_size]
        ratio = SequenceMatcher(None, tsv_text, sub_str).ratio()
        if ratio > best_ratio:
            best_ratio = ratio
            if best_ratio >= 0.80:
                return True
    return best_ratio >= min_ratio

def is_single_block_match(tsv_sentence, block):
    if SequenceMatcher(None, tsv_sentence, block['text']).ratio() >= MATCH_THRESHOLD:
        return True
    if tsv_sentence in block['text'] or block['text'] in tsv_sentence:
        return True
    if window_search_match(tsv_sentence, block['text'], min_ratio=0.50):
        return True
    return False

def parse_ass_file_for_audio(ass_path):
    temp_blocks = {}
    if not os.path.exists(ass_path):
        return []

    with open(ass_path, 'r', encoding='utf-8', errors='ignore') as f:
        for line in f:
            if line.startswith("Dialogue:"):
                parts = line.split(",", 9)
                if len(parts) == 10:
                    start_str = parts[1]
                    end_str = parts[2]
                    style = parts[3].strip()
                    text_content = parts[9]
                    
                    if style.lower() == 'text' and '●' in text_content:
                        start_ms = ass_time_to_ms(start_str)
                        end_ms = ass_time_to_ms(end_str)
                        
                        if start_ms is not None and end_ms is not None:
                            time_key = (start_ms, end_ms)
                            if time_key in temp_blocks:
                                temp_blocks[time_key] += " " + text_content
                            else:
                                temp_blocks[time_key] = text_content
    
    blocks = []
    for (start_ms, end_ms), raw_text in temp_blocks.items():
        blocks.append({
            'start': start_ms,
            'end': end_ms,
            'text': normalize_text_audio(raw_text)
        })
    blocks.sort(key=lambda x: x['start'])
    return blocks

        generated_audios_cache = {}

            start_time = found_block['start']
            end_time = found_block['end']
            
            text_hash = hashlib.md5(normalized_tsv_sentence.encode('utf-8')).hexdigest()[:10]
            audio_filename = f"audio_{text_hash}.mp3"

                continue

            intervals = []
            for item in matched_items:

                start_pos = 0
                while True:

                    if idx == -1:

                    intervals.append({

            print_progress_bar(line_idx + 1, total_lines, prefix='Subtitle Progress:', suffix=f'({line_idx + 1}/{total_lines} lines)', length=40)

        print("\nChecking for unmatched subtitle entries...")

                unmatched_subs_count += 1
                print(f" -> Table row {t_idx + 2}: No ASS match for word '{row[WORD_COLUMN_NAME]}' in '{row[SENTENCE_COLUMN_NAME][:20]}...'")

        try:

        with open(ass_path, mode="w", encoding="utf-8") as f:

    # Rename prep input TSV to _processed.tsv
    processed_tsv_path = os.path.join(orig_dir, f"{base_name}_processed.tsv")
    try:

def scan_batch_triples(selected_dir):
    """Scans the directory tree for matching TSV, ASS, and Media triples"""
    valid_triples = []
    orphans = []
    
    media_extensions = ('.m4a', '.mp3', '.wav', '.aac', '.mp4', '.mkv', '.avi', '.webm')

    for root, dirs, files in os.walk(selected_dir):
        tsv_map = {}
        ass_map = {}
        media_map = {}

        for f in files:
            ext = os.path.splitext(f)[1].lower()
            base = os.path.splitext(f)[0]

            if ext in ['.tsv', '.csv']:
                if base.endswith('_processed') or base.endswith('_output') or base.startswith('output_'):
                    continue
                tsv_map[base] = os.path.join(root, f)
            elif ext == '.ass':
                ass_map[base] = os.path.join(root, f)
            elif ext in media_extensions:
                media_map[base] = os.path.join(root, f)

        tsv_set = set(tsv_map.keys())
        ass_set = set(ass_map.keys())
        media_set = set(media_map.keys())

        matched_bases = tsv_set.intersection(ass_set).intersection(media_set)

        for base in matched_bases:
            valid_triples.append({
                'base': base,
                'tsv_path': tsv_map[base],
                'ass_path': ass_map[base],
                'media_path': media_map[base],
                'folder': root
            })

        all_found = tsv_set.union(ass_set).union(media_set)
        for base in all_found - matched_bases:
            missing = []
            if base not in tsv_set: missing.append("TSV")
            if base not in ass_set: missing.append("ASS")
            if base not in media_set: missing.append("Media")
            orphans.append(f"Incomplete matching for '{base}' in '{root}' (Missing: {', '.join(missing)})")

    return valid_triples, orphans

# --- ROTATING BACKUP SYSTEM (MAX 20) ---

def create_injection_backup(notes, target_fields):
    backup_dir = get_backup_dir()
    existing_backups = sorted(glob.glob(os.path.join(backup_dir, "backup_*.tsv")))
    while len(existing_backups) >= 20:
        try:
            os.remove(existing_backups[0])
            existing_backups.pop(0)
        except Exception:
            break

    timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    filename = f"backup_{timestamp}.tsv"
    file_path = os.path.join(backup_dir, filename)

    try:
        with open(file_path, mode='w', encoding='utf-8-sig', newline='') as f:
            writer = csv.writer(f, delimiter='\t')
            headers = ["nid", "tags"] + list(target_fields)
            writer.writerow(headers)
            for note in notes:
                row = [str(note.id), " ".join(note.tags)]
                for field in target_fields:
                    row.append(note[field])
                writer.writerow(row)
    except Exception as e:
        print(f"Failed to create backup: {e}")

        fields = [f['name'] for f in model['flds']]

            if dlg.exec():
                name, data = dlg.get_data()
                data["model_name"] = model_name
                self.profiles["injection"][name] = data
                save_profiles(self.profiles)
                self.update_list(self.inj_list, "injection")
                showInfo("Injection Profile created successfully.", self)

def html_to_plain_text(html_str):
    if not html_str or not isinstance(html_str, str): return ""
    text = re.sub(r'<(br\s*/?|div|p|li|tr)[^>]*>', '\n', html_str, flags=re.IGNORECASE)
    text = re.sub(r'<[^>]*>', ' ', text)
    text = html.unescape(text)
    text = text.replace('\xa0', ' ')
    text = re.sub(r'[ \t]+', ' ', text)
    text = re.sub(r'\r', '', text)
    text = re.sub(r'\n\s*\n', '\n\n', text)
    return text.strip()

def run_extraction(browser, profile=None):
    notes, fields = get_selected_notes(browser)
    if not notes: return
    
    if profile:
        if not validate_profile_model(notes, profile['model_name'], browser): return
        selected_fields = profile['selected_fields']
        export_as_html = profile['export_as_html']
        export_nid = profile['export_nid']
        export_tags = profile['export_tags']
        if profile['file_path_mode'] == 'fixed' and profile['fixed_file_path']:
            file_path = profile['fixed_file_path']
        else:
            suggested_file = os.path.join(get_last_path(), "extracted_fields.tsv")

    else:

        if not dlg.exec(): return
        selected_fields = dlg.get_selected_fields()
        if not selected_fields:
            showWarning("No fields selected.", parent=browser)
            return
            
        format_choices = ["1. Save as HTML", "2. Save as Plain Text"]
        choice, ok = QInputDialog.getItem(browser, "Export Format", "How should it be exported?", format_choices, 0, False)
        if not ok: return
        export_as_html = (choice == format_choices[0])
        export_nid = (QMessageBox.question(browser, "Export NID?", "Export 'nid' column?", QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No) == QMessageBox.StandardButton.Yes)
        export_tags = (QMessageBox.question(browser, "Export Tags?", "Export 'tags' column?", QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No) == QMessageBox.StandardButton.Yes)
        
        suggested_file = os.path.join(get_last_path(), "extracted_fields.tsv")
        file_path, _ = QFileDialog.getSaveFileName(browser, "Save TSV File", suggested_file, "TSV Files (*.tsv);;Text Files (*.txt)")
        if not file_path: return
        save_last_path(file_path)

        with open(file_path, mode='w', encoding='utf-8-sig', newline='') as f:

            if export_tags: headers.append("tags")
            headers.extend(selected_fields)

    except Exception as e:
        showWarning(f"Error during export: {str(e)}", parent=browser)


def run_injection(browser, profile=None):
    notes, fields = get_selected_notes(browser)
    if not notes: return
    
    if profile:
        if not validate_profile_model(notes, profile['model_name'], browser): return
        mode = profile['injection_mode']
        import_as_html = profile['import_as_html']
        target_fields = profile['target_fields']
        use_nid_matching = profile['use_nid_matching']
        
        tag_handling = profile.get("tag_handling")
        if tag_handling is None:
            has_tags_col = profile.get("has_tags_col", False)
            tag_handling = "overwrite" if has_tags_col else "ignore"
            
        field_mapping = profile['field_mapping']
        
        if profile['file_path_mode'] == 'fixed' and profile['fixed_file_path']:
            file_path = profile['fixed_file_path']
        else:
            file_path, _ = QFileDialog.getOpenFileName(browser, "Open TSV File", get_last_path())
            if not file_path: return
            save_last_path(file_path)
            
    else:
        mode_choices = ["Replace", "Prepend", "Append"]
        mode, ok = QInputDialog.getItem(browser, "Injection Mode", "Choose how to inject the data:", mode_choices, 0, False)
        if not ok: return
            
        format_choices = ["1. Import as HTML (will be rendered)", "2. Import as Plain Text (will be escaped)"]
        choice, ok = QInputDialog.getItem(browser, "Import Format", "How should the text be handled?", format_choices, 0, False)
        if not ok: return
        import_as_html = (choice == format_choices[0])

        field_dlg = FieldSelectionDialog(fields, browser, title="Injection - Select Target Fields")
        if not field_dlg.exec(): return
        target_fields = field_dlg.get_selected_fields()
        if not target_fields:
            showWarning("No target fields selected.", parent=browser)
            return

        file_path, _ = QFileDialog.getOpenFileName(browser, "Open TSV File", get_last_path())
        if not file_path: return
        save_last_path(file_path)

    try:
        with open(file_path, mode='r', encoding='utf-8-sig') as f:
            reader = csv.reader(f, delimiter='\t')
            headers = next(reader, None)
            if not headers:
                showWarning("TSV file is empty.", parent=browser)
                return
            
            headers = [h.strip() for h in headers]
            
            if not profile:
                use_nid_matching = False
                if "nid" in headers:
                    if QMessageBox.question(browser, "NID Matching", "The 'nid' column was found. Match exactly via Note ID (NID)?", QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No) == QMessageBox.StandardButton.Yes:
                        use_nid_matching = True
                
                tag_handling = "ignore"
                if "tags" in headers:
                    tag_choices = [
                        "A) Overwrite existing tags",
                        "B) Append to existing tags",
                        "C) Ignore tags"
                    ]
                    tag_choice, ok = QInputDialog.getItem(
                        browser, 
                        "Tag Handling", 
                        "The 'tags' column was found.\nHow should tags be handled?", 
                        tag_choices, 
                        0, 
                        False
                    )
                    if not ok:
                        return
                    
                    if "Overwrite" in tag_choice:
                        tag_handling = "overwrite"
                    elif "Append" in tag_choice:
                        tag_handling = "append"
                    else:
                        tag_handling = "ignore"
            
            tsv_rows = []
            tsv_data_map = {}
            for row in reader:
                if not row: continue
                if len(row) < len(headers): row += [""] * (len(headers) - len(row))
                row_dict = dict(zip(headers, row))
                tsv_rows.append(row_dict)
                if use_nid_matching and "nid" in row_dict:
                    tsv_data_map[row_dict["nid"].strip()] = row_dict
                    
    except Exception as e:
        showWarning(f"Failed to read file: {str(e)}", parent=browser)
        return

    if not profile:
        mapping_headers = [h for h in headers if h not in ("nid", "tags")]
        map_dlg = MappingDialog(mapping_headers, target_fields, browser)
        if not map_dlg.exec(): return
        field_mapping = map_dlg.get_mapping()
        if not field_mapping and tag_handling == "ignore":
            showWarning("No fields were mapped and tags are ignored. Injection cancelled.", parent=browser)
            return

    create_injection_backup(notes, target_fields)

    def do_injection_op(col):
        updated_notes = []
        for idx, note in enumerate(notes):
            row_dict = None
            if use_nid_matching:
                nid_str = str(note.id)
                if nid_str in tsv_data_map:
                    row_dict = tsv_data_map[nid_str]
            else:
                if idx < len(tsv_rows): row_dict = tsv_rows[idx]
                else: break
                    
            if row_dict:
                if tag_handling != "ignore" and "tags" in row_dict:
                    raw_tags = row_dict.get("tags", "")
                    new_tags_list = col.tags.split(raw_tags)
                    
                    if tag_handling == "overwrite":
                        note.tags = new_tags_list
                    elif tag_handling == "append":
                        combined_tags = list(set(note.tags + new_tags_list))
                        note.tags = [t for t in combined_tags if t.strip()]

                for anki_field, tsv_col in field_mapping.items():
                    if tsv_col not in row_dict: continue
                    new_value = row_dict[tsv_col]
                    if not import_as_html:
                        new_value = html.escape(new_value).replace('\n', '<br>')
                    
                    if mode == "Replace": note[anki_field] = new_value
                    elif mode == "Prepend": note[anki_field] = new_value + note[anki_field]
                    elif mode == "Append": note[anki_field] = note[anki_field] + new_value
                        
                updated_notes.append(note)
                
        changes = col.update_notes(updated_notes)
        return InjectionResult(len(updated_notes), changes)

    def on_success(result):
        browser.model.reset()
        mw.reset()
        showInfo(
            f"Successfully updated {result.count} notes.\n\n"
            "If there are errors, select 'Field_Extract_Inject: Undo Field Injection...' from the Edit menu.",
            parent=browser
        )

    op = CollectionOp(parent=browser, op=do_injection_op)
    op.success(on_success)
    op.run_in_background()

def run_undo_injection(browser):
    backup_dir = get_backup_dir()
    backups = glob.glob(os.path.join(backup_dir, "backup_*.tsv"))
    if not backups:
        showInfo("No Field Injection backups found.", parent=browser)
        return
        
    dlg = BackupSelectionDialog(backups, browser)
    if not dlg.exec(): return
    selected_backup = dlg.get_selected_backup()
    if not selected_backup: return
        
    try:
        with open(selected_backup, mode='r', encoding='utf-8-sig') as f:
            reader = csv.reader(f, delimiter='\t')
            headers = next(reader, None)
            if not headers or "nid" not in headers:
                showWarning("Invalid backup file format.", parent=browser)
                return
            backup_data = {}
            for row in reader:
                if not row: continue
                if len(row) < len(headers): row += [""] * (len(headers) - len(row))
                row_dict = dict(zip(headers, row))
                backup_data[row_dict["nid"].strip()] = row_dict
    except Exception as e:
        showWarning(f"Failed to read backup file: {e}", parent=browser)
        return

    fields_to_restore = [h for h in headers if h not in ("nid", "tags")]
    notes_to_backup = []
    for nid_str in backup_data.keys():
        try: notes_to_backup.append(mw.col.get_note(int(nid_str)))
        except: continue

    if not notes_to_backup:
        showWarning("None of the notes from the backup could be found in your collection.", parent=browser)
        return

    create_injection_backup(notes_to_backup, fields_to_restore)

    def do_restore_op(col):
        restored_notes = []
        for note in notes_to_backup:
            nid_str = str(note.id)
            if nid_str in backup_data:
                row_dict = backup_data[nid_str]
                if "tags" in row_dict: note.tags = col.tags.split(row_dict["tags"])
                for field in fields_to_restore:
                    if field in note: note[field] = row_dict[field]
                restored_notes.append(note)
        changes = col.update_notes(restored_notes)
        return InjectionResult(len(restored_notes), changes)

    def on_restore_success(result):
        browser.model.reset()
        mw.reset()
        tooltip(f"<b>Restored</b> {result.count} notes from backup.", parent=browser)

    op = CollectionOp(parent=browser, op=do_restore_op)
    op.success(on_restore_success)
    op.run_in_background()

def run_manual_linker(browser):
    # Get all selected card IDs
    selected_cids = browser.selectedCards()
    
    if len(selected_cids) < 2:
        showInfo("Please select at least two cards to link them together.")
        return

    mw.checkpoint("Cross-Linker Run")
    mw.progress.start()
    
    debug_log = []
    success_count = 0

    try:
        # Collect note data and labels for all selected cards first
        notes_data = []
        for cid in selected_cids:
            try:
                card = mw.col.get_card(cid)
                note = card.note()
                if TARGET_FIELD not in note:
                    debug_log.append(f"⚠️ Field '{TARGET_FIELD}' missing on note ID {note.id}. Skipping this note.")
                    continue
                
                label = note[SOURCE_LABEL_FIELD].strip()
                notes_data.append({
                    "note": note,
                    "id": note.id,
                    "label": label
                })
            except Exception as e:
                debug_log.append(f"❌ Error reading card ID {cid}: {str(e)}")

        # Timestamp for the connection block
        timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M")

        # Update each note with the links of all OTHER notes in the selection
        for current in notes_data:
            link_entries = []
            
            for other in notes_data:
                # Link every note except the current one itself
                if current["id"] != other["id"]:
                    link_entries.append(f"[{other['label']}|nid{other['id']}]")
            
            if link_entries:
                current_note = current["note"]
                
                # Format the new content block
                new_links_block = f"Cross-Link Connection ({timestamp})<br>" + "<br>".join(link_entries)
                
                # FETCH EXISTING CONTENT AND APPEND Safely
                existing_content = current_note[TARGET_FIELD].strip()
                if existing_content:
                    # If there is already text, add a double break and append the new links
                    current_note[TARGET_FIELD] = f"{existing_content}<br><br>{new_links_block}"
                else:
                    # If the field was empty, just insert the new links
                    current_note[TARGET_FIELD] = new_links_block
                
                mw.col.update_note(current_note)
                success_count += 1

    except Exception:
        debug_log.append(f"💥 Critical system error:\n{traceback.format_exc()}")

    finally:
        mw.progress.finish()
        browser.model.reset()
        mw.reset()

    if debug_log:
        showInfo(f"Process finished with messages/errors:\n\n" + "\n".join(debug_log))
    else:
        tooltip(f"Success! {success_count} cards cross-linked with each other (Appended).")

def setup_menu(browser):
    # New menu item for the Cross-Linker
    action = QAction("Cross-Linker (Link All Selected Together)", browser)
    action.setShortcut(QKeySequence("Ctrl+Alt+M"))
    action.triggered.connect(lambda: run_manual_linker(browser))
    browser.form.menuEdit.addAction(action)

gui_hooks.browser_menus_did_init.append(setup_menu)

def clean_field(text):
    """Helper function to clean CSV/TSV strings (Removes BOM and quotes)"""
    if not text:
        return ""
    text = text.strip().replace('\ufeff', '')
    if text.startswith('"') and text.endswith('"'):
        text = text[1:-1].replace('""', '"')
    return text

def html_to_plain_text(html_str):
    if not html_str or not isinstance(html_str, str): return ""
    text = re.sub(r'<(br\s*/?|div|p|li|tr)[^>]*>', '\n', html_str, flags=re.IGNORECASE)
    text = re.sub(r'<[^>]*>', ' ', text)
    text = html.unescape(text)
    text = text.replace('\xa0', ' ')
    text = re.sub(r'[ \t]+', ' ', text)
    text = re.sub(r'\r', '', text)
    text = re.sub(r'\n\s*\n', '\n\n', text)
    return text.strip()

def start_mining_process():
    # 1. FILE SELECTION VIA FILE EXPLORER
    indiv_file = getFile(mw, "Select the file for INDIVIDUAL CARDS (anki_individual_cards.tsv)", None, "TSV Files (*.tsv);;All Files (*)")
    if not indiv_file:
        return
    
    update_file = getFile(mw, "Select the file for GRAMMAR UPDATES (anki_grammar_updates.tsv)", None, "TSV Files (*.tsv);;All Files (*)")
    if not update_file:
        return

    mw.checkpoint("Grammar Mining and Linking")
    mw.progress.start(immediate=True)

    try:
        deck_name = "mining"
        model_name = "miningsimple"
        
        deck_id = mw.col.decks.id(deck_name)
        model = mw.col.models.by_name(model_name)
        
        if not model:
            mw.progress.finish()
            showCritical(f"The note type '{model_name}' was not found! Please check the exact spelling in Anki.")
            return
            
        mw.col.decks.select(deck_id)
        model['did'] = deck_id
        mw.col.models.setCurrent(model)

        anki_fields = {f['name']: f['ord'] for f in model['flds']}
        indiv_id_to_anki_nid = {}
        new_note_ids = []

        with open(indiv_file, 'r', encoding='utf-8-sig') as f:
            reader = csv.reader(f, delimiter='\t')
            headers = [clean_field(h) for h in next(reader)] 
            
            if "Note ID" not in headers:
                mw.progress.finish()
                showCritical("The individual file does not have a 'Note ID' header!")
                return

            for row in reader:
                if not row or len(row) < len(headers):
                    continue
                
                row_data = {headers[i]: clean_field(row[i]) for i in range(len(row))}
                
                if row_data["Note ID"] == "Note ID" or row_data["Note ID"] == "nidindiv":
                    continue
                
                note = mw.col.new_note(model)
                
                for col_name, val in row_data.items():
                    if col_name in anki_fields:
                        note.fields[anki_fields[col_name]] = val
                
                note.add_tag("grammar")
                mw.col.add_note(note, deck_id)
                new_note_ids.append(note.id)
                
                nidindiv = row_data["Note ID"]
                indiv_id_to_anki_nid[nidindiv] = {
                    "anki_nid": note.id,
                    "term_content": row_data.get("Word", "")
                }

        for nid in new_note_ids:
            note = mw.col.get_note(nid)
            for card in note.cards():
                if "pronounciation" in card.template()['name'].lower():
                    card.queue = -1  
                    card.flush()

        grammar_points_cache = {}

        with open(update_file, 'r', encoding='utf-8-sig') as f:
            reader = csv.reader(f, delimiter='\t')
            update_headers = [clean_field(h) for h in next(reader)]
            
            for row in reader:
                if not row or len(row) < 2:
                    continue
                    
                grammar_nid = clean_field(row[0])
                collected_sentences_raw = clean_field(row[1])
                
                if grammar_nid == "nid":
                    continue
                
                try:
                    grammar_note = mw.col.get_note(int(grammar_nid))
                except Exception:
                    continue

                g_fields = {f['name']: f['ord'] for f in grammar_note.model()['flds']}
                
                if "mined sentences" not in g_fields or "Level And Grammar Point" not in g_fields:
                    continue
                
                grammar_point_b = grammar_note.fields[g_fields["Level And Grammar Point"]].strip()
                grammar_points_cache[grammar_nid] = grammar_point_b

                sentences = collected_sentences_raw.split("<br>")
                processed_sentences = []

                for sentence in sentences:
                    sentence = sentence.strip()
                    if not sentence:
                        continue
                    
                    match = re.search(r'\[([A-Za-z0-9\-]+)\]$', sentence)
                    if match:
                        nidindiv = match.group(1)
                        
                        if nidindiv in indiv_id_to_anki_nid:
                            nid_a = indiv_id_to_anki_nid[nidindiv]["anki_nid"]
                            sentence_c = sentence[:match.start()].strip()
                            
                            linked_sentence = f"[{sentence_c}|nid{nid_a}]"
                            processed_sentences.append(linked_sentence)
                            
                            try:
                                indiv_note = mw.col.get_note(nid_a)
                                if "Frequency" in anki_fields:
                                    current_val = indiv_note.fields[anki_fields["Frequency"]].strip()
                                    backlink = f"[{grammar_point_b}|nid{grammar_nid}]"
                                    if current_val:
                                        indiv_note.fields[anki_fields["Frequency"]] = f"{current_val}<br>{backlink}"
                                    else:
                                        indiv_note.fields[anki_fields["Frequency"]] = backlink
                                        
                                    indiv_note.flush()
                            except Exception as e:
                                print(f"Error backlinking note {nid_a}: {e}")
                        else:
                            processed_sentences.append(sentence)
                    else:
                        processed_sentences.append(sentence)

                current_mined_sentences = grammar_note.fields[g_fields["mined sentences"]].strip()
                new_sentences_block = "<br>".join(processed_sentences)
                
                if current_mined_sentences:
                    grammar_note.fields[g_fields["mined sentences"]] = f"{new_sentences_block}<br>{current_mined_sentences}"
                else:
                    grammar_note.fields[g_fields["mined sentences"]] = new_sentences_block
                
                grammar_note.flush()

        mw.progress.finish()
        mw.reset()
        showInfo("Mining, suspension, and link generation completed successfully!")

    except Exception as e:
        mw.progress.finish()
        showCritical(f"A critical error occurred:\n{str(e)}")

    def browse_script(self):
        path, _ = QFileDialog.getOpenFileName(self, "Select Python Script", os.path.expanduser("~"), "Python Scripts (*.py)")
        if path:
            self.script_input.setText(path)

    def save_current_filter(self):
        current_filter = self.filter_input.text().strip()
        self.cfg["search_filter"] = current_filter
        save_gdu_config(self.cfg)
        showInfo("Search filter was successfully saved in the configuration.", self)

    def run_operation(self):
        search_filter = self.filter_input.text().strip()
        if not search_filter:
            showWarning("Search filter cannot be empty.", self)
            return
            
        script_mode = "fixed" if self.mode_combo.currentIndex() == 1 else "variable"
        script_path = self.script_input.text().strip()
        
        if script_mode == "fixed" and not script_path:
            showWarning("Please specify a path to the script.", self)
            return
            
        if script_mode == "variable":
            path, _ = QFileDialog.getOpenFileName(self, "Select Python Script to Execute", os.path.expanduser("~"), "Python Scripts (*.py)")
            if not path:
                return 
            script_path = path

        # Find all cards using the filter (no start NID)
        all_matching_nids = mw.col.find_notes(search_filter)
        
        if not all_matching_nids:
            showInfo("No cards found for the specified filter.", self)
            return
            
        ext_profile = get_extraction_profile("GrammarDataUpdate")
        if not ext_profile:
            showWarning("Extraction profile 'GrammarDataUpdate' was not found in the Field_Extract_Inject configuration.", self)
            return
            
        try:
            selected_fields = ext_profile.get("selected_fields", [])
            export_as_html = ext_profile.get("export_as_html", False)
            export_nid = ext_profile.get("export_nid", False)
            export_tags = ext_profile.get("export_tags", False)
            
            if ext_profile.get("file_path_mode") == "fixed" and ext_profile.get("fixed_file_path"):
                out_path = ext_profile["fixed_file_path"]
            else:
                out_path, _ = QFileDialog.getSaveFileName(self, "Save TSV File", os.path.expanduser("~"), "TSV Files (*.tsv)")
                if not out_path:
                    return

            notes = [mw.col.get_note(nid) for nid in all_matching_nids]
            
            with open(out_path, mode='w', encoding='utf-8-sig', newline='') as f:
                writer = csv.writer(f, delimiter='\t')
                headers = []
                if export_nid: headers.append("nid")
                if export_tags: headers.append("tags")
                headers.extend(selected_fields)
                writer.writerow(headers)
                
                for note in notes:
                    row = []
                    if export_nid: row.append(str(note.id))
                    if export_tags: row.append(" ".join(note.tags))
                    for field in selected_fields:
                        val = note[field] if field in note else ""
                        if not export_as_html: val = html_to_plain_text(val)
                        row.append(val)
                    writer.writerow(row)
                    
        except Exception as e:
            showWarning(f"Error during extraction: {str(e)}", self)
            return
            
        # Update configuration at the end
        self.cfg["search_filter"] = search_filter
        self.cfg["script_mode"] = script_mode
        self.cfg["script_path"] = script_path if script_mode == "fixed" else self.script_input.text()
        save_gdu_config(self.cfg)
        
        try:
            script_dir = os.path.dirname(os.path.abspath(script_path))
            
            subprocess.Popen(
                ["python", script_path], 
                shell=True, 
                cwd=script_dir
            )
            
            showInfo(f"Successfully extracted {len(all_matching_nids)} cards.\n\nTSV exported to:\n{out_path}\n\nExternal script has been started.", self)
            self.accept()
        except Exception as e:
            showWarning(f"Extraction successful, but the external script could not be started:\n{str(e)}", self)

    def run_operation(self):
        try:
            start_nid = int(self.nid_input.text().strip())
        except ValueError:
            showWarning("Start Point must be a valid numeric Note ID.", self)
            return
            
        search_filter = self.filter_input.text().strip()
        if not search_filter:
            showWarning("Search Filter cannot be empty.", self)
            return
            
        script_mode = "fixed" if self.mode_combo.currentIndex() == 1 else "variable"
        script_path = self.script_input.text().strip()
        
        if script_mode == "fixed" and not script_path:
            showWarning("Please specify a fixed script path.", self)
            return
            
        if script_mode == "variable":
            path, _ = QFileDialog.getOpenFileName(self, "Select Python Script to Execute", os.path.expanduser("~"), "Python Scripts (*.py)")
            if not path:
                return 
            script_path = path

        all_matching_nids = mw.col.find_notes(search_filter)
        new_nids = [nid for nid in all_matching_nids if nid > start_nid]
        new_nids.sort() 
        
        if not new_nids:
            showInfo("No new notes found past the starting point.", self)
            return
            
        ext_profile = get_extraction_profile("AddNewVocToList")
        if not ext_profile:
            showWarning("Extraction profile 'AddNewVocToList' not found in Field_Extract_Inject config.", self)
            return
            
        try:
            selected_fields = ext_profile.get("selected_fields", [])
            export_as_html = ext_profile.get("export_as_html", False)
            export_nid = ext_profile.get("export_nid", False)
            export_tags = ext_profile.get("export_tags", False)
            
            if ext_profile.get("file_path_mode") == "fixed" and ext_profile.get("fixed_file_path"):
                out_path = ext_profile["fixed_file_path"]
            else:
                out_path, _ = QFileDialog.getSaveFileName(self, "Save TSV File", os.path.expanduser("~"), "TSV Files (*.tsv)")
                if not out_path:
                    return

            notes = [mw.col.get_note(nid) for nid in new_nids]
            
            with open(out_path, mode='w', encoding='utf-8-sig', newline='') as f:
                writer = csv.writer(f, delimiter='\t')
                headers = []
                if export_nid: headers.append("nid")
                if export_tags: headers.append("tags")
                headers.extend(selected_fields)
                writer.writerow(headers)
                
                for note in notes:
                    row = []
                    if export_nid: row.append(str(note.id))
                    if export_tags: row.append(" ".join(note.tags))
                    for field in selected_fields:
                        val = note[field] if field in note else ""
                        if not export_as_html: val = html_to_plain_text(val)
                        row.append(val)
                    writer.writerow(row)
                    
        except Exception as e:
            showWarning(f"Error during extraction: {str(e)}", self)
            return
            
        highest_nid = new_nids[-1]
        
        # Speichere die aktuellsten Werte am Ende der Operation ab
        self.cfg["start_nid"] = highest_nid
        self.cfg["search_filter"] = search_filter
        self.cfg["script_mode"] = script_mode
        self.cfg["script_path"] = script_path if script_mode == "fixed" else self.script_input.text()
        save_config(self.cfg)
        
        self.nid_input.setText(str(highest_nid))
        
        try:
            # Ermittle den Ordner, in dem das Skript liegt, für das Arbeitsverzeichnis (cwd)
            script_dir = os.path.dirname(os.path.abspath(script_path))
            
            # Unter Windows ist es oft sicherer, shell=True zu nutzen, 
            # damit die Windows-Umgebungsvariablen für "python" greifen.
            # Zudem setzen wir cwd, damit das Skript seine relativen Pfade (z.B. den "data"-Ordner) findet!
            subprocess.Popen(
                ["python", script_path], 
                shell=True, 
                cwd=script_dir
            )
            
            showInfo(f"Successfully processed {len(new_nids)} notes.\n\nTSV exported to:\n{out_path}\n\nExternal script launched.", self)
            self.accept()
        except Exception as e:
            showWarning(f"Extraction successful, but failed to launch external script:\n{str(e)}", self)

# -*- coding: utf-8 -*-

import re
from aqt import mw, gui_hooks
from aqt.qt import QAction, QKeySequence, QInputDialog
from aqt.utils import tooltip

# --- CONFIGURATION ---
TARGET_FIELD_LINK = "Link to Related Cards"
SOURCE_FIELD_LINK = "Word"
FIELDS_TO_PROCESS = ["SentencePlain", "SentenceFurigana"]
GROUP_CRITERION_FIELD = "SentencePlain"
# ---------------------

def clean_html(raw_html):
    """Removes all HTML tags to compare the plain text."""
    clean_r = re.compile('<.*?>')
    return re.sub(clean_r, '', raw_html).strip()

def build_link_string(notes, prefix):
    """Builds the exact link string in the format [Word|nid12345]."""
    parts = []
    for note in reversed(notes):
        word = note[SOURCE_FIELD_LINK].strip() if SOURCE_FIELD_LINK in note and note[SOURCE_FIELD_LINK] else "(no word)"
        parts.append(f"[{word}|nid{note.id}]")
    return f"{prefix}: " + " ".join(parts)

def remove_link_line(note, target_link_string):
    """Removes a specific link line from the target field of the note."""
    if TARGET_FIELD_LINK not in note:
        return False
    current_content = note[TARGET_FIELD_LINK].strip()
    if not current_content:
        return False
    lines = [line.strip() for line in re.split(r'<br\s*/?>', current_content)]
    new_lines = [line for line in lines if line != target_link_string]
    if len(lines) != len(new_lines):
        note[TARGET_FIELD_LINK] = "<br>".join(new_lines)
        return True
    return False

# FUNCTION: RUN LINK OPERATIONS (DO)
# ==========================================
def run_combined_operations(browser):
    cids = browser.selectedCards()
    if not cids:
        tooltip("No cards selected.")
        return

    options = ["1) sentence match link", "2) batch link"]
    item, ok = QInputDialog.getItem(browser, "Select Mode", "How should the cards be processed?", options, 0, False)
    if not ok or not item:
        return

    mode = 1 if item.startswith("1") else 2
    mw.checkpoint("Batch Link Operations")
    mw.progress.start()
    
    total_linked = 0
    total_highlights = 0
    processed_batches = 0

    try:
        if mode == 1:
            groups = {}
            for cid in cids:
                note = mw.col.get_card(cid).note()
                if GROUP_CRITERION_FIELD in note:
                    cleaned_content = clean_html(note[GROUP_CRITERION_FIELD])
                    if cleaned_content:
                        if cleaned_content not in groups:
                            groups[cleaned_content] = []
                        if note.id not in [n.id for n in groups[cleaned_content]]:
                            groups[cleaned_content].append(note)

            for cleaned_sentence, batch_notes in groups.items():
                if len(batch_notes) > 1:
                    processed_batches += 1
                    new_link_content = "[Sentence Link]: " + build_link_string(batch_notes, "").replace(": ", "", 1)
                    
                    # Collect highlights
                    all_bold_words = set()
                    for note in batch_notes:
                        for field in FIELDS_TO_PROCESS:
                            if field in note:
                                for word in re.findall(r'<b>(.*?)</b>', note[field]):
                                    if word.strip():
                                        clean_w = re.sub(r'\[.*?\]', '', word)
                                        all_bold_words.add(clean_w)
                                        if word != clean_w:
                                            all_bold_words.add(word)
                    sorted_highlights = sorted(list(all_bold_words), key=len, reverse=True)

                    for note in batch_notes:
                        changed = False
                        if TARGET_FIELD_LINK in note:
                            current = note[TARGET_FIELD_LINK].strip()
                            note[TARGET_FIELD_LINK] = (new_link_content + "<br>" + current) if current else new_link_content
                            changed = True
                            total_linked += 1

                        for field in FIELDS_TO_PROCESS:
                            if field in note:
                                orig = note[field]
                                txt = re.sub(r'<u>(.*?)</u>', r'\1', orig)
                                txt = re.sub(r'<i>(.*?)</i>', r'\1', txt)
                                current_bold = re.findall(r'<b>(.*?)</b>', txt)
                                for word in sorted_highlights:
                                    if word in current_bold:
                                        txt = txt.replace(f"<b>{word}</b>", f"<b><u>{word}</u></b>")
                                    elif word in txt and f"<u>{word}</u>" not in txt:
                                        txt = re.sub(f'(?<!<)(?<!/){re.escape(word)}(?!>)', f'<u>{word}</u>', txt)
                                if txt != orig:
                                    note[field] = txt
                                    changed = True
                                    total_highlights += 1
                        if changed:
                            mw.col.update_note(note)

        elif mode == 2:
            processed_batches = 1
            seen_nids = set()
            unique_notes = []
            for cid in cids:
                note = mw.col.get_card(cid).note()
                if note.id not in seen_nids:
                    seen_nids.add(note.id)
                    unique_notes.append(note)
            
            new_link_content = "[Batch Link]: " + build_link_string(unique_notes, "").replace(": ", "", 1)
            for note in unique_notes:
                if TARGET_FIELD_LINK in note:
                    current = note[TARGET_FIELD_LINK].strip()
                    note[TARGET_FIELD_LINK] = (new_link_content + "<br>" + current) if current else new_link_content
                    mw.col.update_note(note)
                    total_linked += 1

    finally:
        mw.progress.finish()
        browser.model.reset()
        mw.reset()

    tooltip(f"Created -> Links: {total_linked} | Highlights: {total_highlights}")

# FUNCTION: RUN UNDO OPERATIONS (UNDO)
# ==========================================
def run_undo_operations(browser):
    cids = browser.selectedCards()
    if not cids:
        tooltip("No cards selected.")
        return

    mw.checkpoint("Batch Link Undo Operations")
    mw.progress.start()

    seen_nids = set()
    notes = []
    for cid in cids:
        note = mw.col.get_card(cid).note()
        if note.id not in seen_nids:
            seen_nids.add(note.id)
            notes.append(note)

    sentence_groups = {}
    for note in notes:
        if GROUP_CRITERION_FIELD in note:
            cleaned = clean_html(note[GROUP_CRITERION_FIELD])
            if cleaned:
                if cleaned not in sentence_groups:
                    sentence_groups[cleaned] = []
                sentence_groups[cleaned].append(note)

    total_links_removed = 0
    total_highlights_removed = 0

    try:
        has_sentence_match = any(len(group) > 1 for group in sentence_groups.values())

        if has_sentence_match:
            for cleaned_sentence, group_notes in sentence_groups.items():
                if len(group_notes) > 1:
                    expected_link = build_link_string(group_notes, "[Sentence Link]")
                    all_bold_words = set()
                    for note in group_notes:
                        for field in FIELDS_TO_PROCESS:
                            if field in note:
                                for word in re.findall(r'<b>(.*?)</b>', note[field]):
                                    if word.strip():
                                        clean_w = re.sub(r'\[.*?\]', '', word)
                                        all_bold_words.add(clean_w)
                                        if word != clean_w:
                                            all_bold_words.add(word)

                    for note in group_notes:
                        changed = False
                        if remove_link_line(note, expected_link):
                            changed = True
                            total_links_removed += 1
                        
                        for field in FIELDS_TO_PROCESS:
                            if field in note:
                                orig = note[field]
                                txt = orig
                                for word in all_bold_words:
                                    txt = txt.replace(f"<b><u>{word}</u></b>", f"<b>{word}</b>")
                                    txt = txt.replace(f"<u>{word}</u>", word)
                                if txt != orig:
                                    note[field] = txt
                                    changed = True
                                    total_highlights_removed += 1
                        if changed:
                            mw.col.update_note(note)
        else:
            expected_link = build_link_string(notes, "[Batch Link]")
            for note in notes:
                if remove_link_line(note, expected_link):
                    total_links_removed += 1
                    mw.col.update_note(note)

    finally:
        mw.progress.finish()
        browser.model.reset()
        mw.reset()

    tooltip(f"Removed -> Links: {total_links_removed} | Highlights: {total_highlights_removed}")

import json

# Target sentences mapped to keys
SEARCH_TARGETS = {
    "こと": ["俺のほかには誰もいない静かなキッチン"],
    "ところ": ["小町が満足げに二人を眺める"],
    "もの": ["打ち上げれば落ちるのは自然の摂理であり"],
    "上": ["健康上ではポテトフライや揚げ物などは食べない方がいいと思います"],

        "路上で歌たい続けようと思います",
    ],

def run_grammar_nid_extractor():
    # Step 1: Prompt for Grammar Deck Name
    deck_name = "grammar"
    is_changed = askUser(
        'Is your grammar deck still named "grammar"?\n\n'
        'Click "Yes" to keep "grammar", or "No" to enter a different deck name.',
        defaultno=False,
    )

    if not is_changed:
        custom_name, ok = getText("Enter your Grammar Deck name:", default="grammar")
        if not ok or not custom_name.strip():
            showCritical("Operation aborted: Invalid deck name.")
            return
        deck_name = custom_name.strip()

    # Step 2: Locate config2.txt file
    showInfo("Please select the target 'config2.txt' file in the next dialog.")
    cfg_path, _ = QFileDialog.getOpenFileName(
        mw,
        "Select or Locate config2.txt",
        "",
        "Text Files (config2.txt *.txt);;All Files (*)",
    )

    if not cfg_path:
        showCritical("Operation aborted: No file selected.")
        return

    # Step 3: Execute note searches
    collected_nids = {}

    for key, sentences in SEARCH_TARGETS.items():
        nids_for_key = []
        for sentence in sentences:
            # Query format: deck:"<name>" "<sentence>"
            query = f'deck:"{deck_name}" "{sentence}"'
            found_nids = mw.col.find_notes(query)

            # Strict single result check
            if len(found_nids) != 1:
                showCritical(
                    f"there should only be one result\n\n"
                    f"Query: {query}\n"
                    f"Found: {len(found_nids)} results."
                )
                return

            nids_for_key.append(str(found_nids[0]))

        # Store single string NID or list for '限り'
        if key == "限り":
            collected_nids[key] = nids_for_key
        else:
            collected_nids[key] = nids_for_key[0]

    # Step 4: Write to config2.txt
    try:
        with open(cfg_path, "w", encoding="utf-8") as f:
            json.dump(collected_nids, f, indent=4, ensure_ascii=False)

        showInfo(
            f"Successfully updated {os.path.basename(cfg_path)} with extracted NIDs!"
        )
    except Exception as e:
        showCritical(f"Failed to write to {cfg_path}:\n{str(e)}")


def setup_menu():
    action = QAction("Extract Grammar NIDs to config2.txt", mw)
    action.triggered.connect(run_grammar_nid_extractor)
    mw.form.menuTools.addAction(action)


setup_menu()

def on_webview_will_set_content(web_content: WebContent, context=None):
    """
    Injiziert JavaScript direkt in den WebView des Reviewers / Previewers.
    """
    if not isinstance(context, (AnkiWebView, object)):
        return

    # JS-Code zur Handhabung von Hover, Popup und Anki-Browser-Suche
    js_code = """
    <script>
    (function() {
        if (window.__sentenceHoverInitialized) return;
        window.__sentenceHoverInitialized = true;

        var hoverTimer = null;
        var isInsidePopup = false;

        window.handleWordHover = function(element) {
            clearTimeout(hoverTimer);
            var popup = document.getElementById('sentence-preview-box');
            var listContainer = document.getElementById('preview-sentences-list');
            var hiddenWordData = document.getElementById('hidden-word-data');

            if (!popup || !hiddenWordData) return;

            popup.style.display = 'block';

            var wordValue = hiddenWordData.innerText.trim();
            if (!wordValue) return;

            if (listContainer) {
                listContainer.innerHTML = '<div style="font-size:14px; opacity:0.6; padding-top:4px;">Lade...</div>';
            }

            // Anki Python Backend nach Sätzen abfragen
            pycmd("get_sentences_for_word:" + encodeURIComponent(wordValue), function(sentences) {
                if (!listContainer) return;

                if (sentences && sentences.length > 0) {
                    var html = '<ul class="preview-list">';
                    sentences.forEach(function(s) {
                        html += '<li>' + s + '</li>';
                    });
                    html += '</ul>';
                    listContainer.innerHTML = html;
                } else {
                    listContainer.innerHTML = '<div style="font-size:14px; opacity:0.6; padding-top:4px;">Keine Sätze gefunden.</div>';
                }
            });
        };

        window.handleWordLeave = function() {
            hoverTimer = setTimeout(function() {
                if (!isInsidePopup) {
                    closePreview();
                }
            }, 200);
        };

        window.keepPreviewOpen = function() {
            isInsidePopup = true;
            clearTimeout(hoverTimer);
        };

        window.closePreview = function() {
            isInsidePopup = false;
            var popup = document.getElementById('sentence-preview-box');
            if (popup) popup.style.display = 'none';
        };

        window.searchWordWithAnkiConnect = function() {
            var hiddenWordData = document.getElementById('hidden-word-data');
            if (!hiddenWordData) return;
            var wordValue = hiddenWordData.innerText.trim();
            pycmd("open_browser_for_word:" + encodeURIComponent(wordValue));
        };
    })();
    </script>
    """

    web_content.head += js_code


def on_pycmd(handled, message, context):
    """
    Verarbeitet Befehle aus dem JavaScript-Code direkt in Python.
    """
    if message.startswith("get_sentences_for_word:"):
        word = urllib.parse.unquote(message.split(":", 1)[1]).strip()

        # Aktuelle Karte ermitteln (sicher für Objekt oder Callable)
        card = getattr(context, "card", None)
        if callable(card):
            card = card()

        if not card:
            return (True, [])

        # Kartentyp-Name ermitteln (z.B. 'term' oder 'pronounciation')
        ord_idx = card.ord
        note_type = card.note_type()
        card_type_name = "term"
        if note_type and ord_idx < len(note_type['tmpls']):
            card_type_name = note_type['tmpls'][ord_idx]['name'].lower()

        # Query für Note-Type miningsimple
        query = f'note:miningsimple card:{card_type_name} -is:suspended Word:"{word}"'

        # Direkte Datenbanksuche über Ankicodes Python API
        note_ids = mw.col.find_notes(query)
        sentences = []

        for nid in note_ids:
            note = mw.col.get_note(nid)
            if "SentencePlain" in note:
                sentence_text = note["SentencePlain"].strip()
                if sentence_text and sentence_text not in sentences:
                    sentences.append(sentence_text)

        return (True, sentences)

    elif message.startswith("open_browser_for_word:"):
        word = urllib.parse.unquote(message.split(":", 1)[1]).strip()

        card = getattr(context, "card", None)
        if callable(card):
            card = card()

        card_type_name = "term"
        if card:
            ord_idx = card.ord
            note_type = card.note_type()
            if note_type and ord_idx < len(note_type['tmpls']):
                card_type_name = note_type['tmpls'][ord_idx]['name'].lower()

        query = f'note:miningsimple card:{card_type_name} -is:suspended Word:"{word}"'

        # Anki Browser über die Qt-Dialoge-API öffnen und durchsuchen
        browser = aqt.dialogs.open("Browser", mw)
        browser.search_for(query)

        return (True, None)

    return handled


# Hooks in Anki registrieren
webview_will_set_content.append(on_webview_will_set_content)
webview_did_receive_js_message.append(on_pycmd)

def nplus1_scan(browser: Browser):
    selected = browser.selectedCards()
    if not selected:
        showInfo("No cards selected.")
        return

    col = mw.col
    notes = [col.get_card(cid).note() for cid in selected]

    # Count SoundFront only among the selected cards
    sound_counts = {}
    for note in notes:
        sf = note["SoundFront"].strip() if "SoundFront" in note else ""
        if sf:
            sound_counts[sf] = sound_counts.get(sf, 0) + 1

    # Keep only cards whose SoundFront is unique among the selected cards
    unique_card_ids = [
        col.get_note(note.id).cards()[0].id  # get the Card ID
        for note in notes
        if "SoundFront" in note and sound_counts[note["SoundFront"].strip()] == 1
    ]

    if not unique_card_ids:
        showInfo("No unique cards found!")
        return

    # Combine IDs with OR
    query = " OR ".join(f"cid:{cid}" for cid in unique_card_ids)
    browser.search_for(query)

    showInfo(f"{len(unique_card_ids)} unique cards listed.")

# --- CONFIGURATION ---
TARGET_FIELD = "Connected Grammar Points from jlptsensei (optional)"
SOURCE_LABEL_FIELD = "Level And Grammar Point"

EXCEPTIONS = {
    "1:に": "1:に–",
    "4:他動詞&nbsp;&amp;自動詞Transitive&amp;IntransitiveVerbs": "4:他動詞",

def set_clipboard(text):
    mode = QApplication.clipboard()
    mode.setText(text)

def extract_connected_points(text):
    """Extracts grammar points from the link format (LOGIC UNCHANGED)."""
    if not text: return []
    pattern = re.compile(r'\[(.*?)\|nid\d+\]')
    matches = pattern.findall(text)
    points = []
    for raw_title in matches:
        clean_title = raw_title.split('(')[0]
        clean_title = clean_title.replace(" / ", "/").replace(" ", "").strip()
        points.append(clean_title)
    return points

def get_prefix(notes):
    combined = ""
    for n in notes:
        label = n[SOURCE_LABEL_FIELD].split('(')[0].strip()
        clean_label = re.sub(r'[^a-zA-Z0-9\u3040-\u309F\u30A0-\u30FF\u4E00-\u9FFF]', '', label)
        combined += clean_label
    prefix = combined[:15]
    if len(prefix) < 5:
        prefix = (prefix + "Group")[:5]
    return prefix

def run_relation_crawler(browser):
    selected_cids = browser.selectedCards()
    if not selected_cids:
        tooltip("No cards selected.")
        return

    # Qt6 Buttons for Anki 25.02.4
    if hasattr(QMessageBox, "StandardButton"):
        BTN_YES = QMessageBox.StandardButton.Yes
        BTN_NO = QMessageBox.StandardButton.No
    else:
        BTN_YES = QMessageBox.Yes
        BTN_NO = QMessageBox.No

    # --- INITIAL USER PROMPTS ---
    degree, ok = QInputDialog.getInt(browser, "Relation Degree", 
                                    "Search Depth (0=selection only, 1=neighbors, etc.):", 
                                    1, 0, 5)
    if not ok: return

    # Ask if tagging should be performed at all
    add_tag_opt = QMessageBox.question(browser, "Add Tags", 
                                       "Should the found cards be tagged?", 
                                       BTN_YES | BTN_NO) == BTN_YES

    final_tag = ""
    initial_notes = [mw.col.get_card(cid).note() for cid in selected_cids]
    tag_prefix = get_prefix(initial_notes)

    if add_tag_opt:
        # Ask if the user wants to customize the tag name
        custom_tag_opt = QMessageBox.question(browser, "Tag Name", 
                                              "Do you want to specify a custom tag name?\n(Click 'No' to use the auto-generated name)", 
                                              BTN_YES | BTN_NO) == BTN_YES
        
        auto_tag = f"degreeRel({degree})_{tag_prefix}"
        if custom_tag_opt:
            custom_tag, ok = QInputDialog.getText(browser, "Custom Tag", 
                                                  "Enter custom tag name:", 
                                                  text=auto_tag)
            if ok and custom_tag.strip():
                final_tag = custom_tag.strip()
            else:
                final_tag = auto_tag # Fallback if canceled or empty
        else:
            final_tag = auto_tag

    # Filtered Deck only makes sense if tagging is enabled
    create_filtered_opt = False
    if add_tag_opt and final_tag:
        create_filtered_opt = QMessageBox.question(browser, "Filtered Deck", 
                                                   "Should a Filtered Deck be created for all tagged cards?", 
                                                   BTN_YES | BTN_NO) == BTN_YES

    # ALWAYS ask for browser overview, regardless of tagging decision
    show_in_browser_opt = QMessageBox.question(browser, "Browser Overview", 
                                                "Show all found cards in the browser after completion?", 
                                                BTN_YES | BTN_NO) == BTN_YES

    show_log_opt = QMessageBox.question(browser, "Execution Log", 
                                        "Display the detailed search log at the end?", 
                                        BTN_YES | BTN_NO) == BTN_YES

    # --- CRAWLER LOGIC ---
    mw.checkpoint("Relation Crawler Update")
    mw.progress.start()

    log = []
    log.append(f"=== Crawler Start (Degree {degree}) ===")

    processed_nids = set(n.id for n in initial_notes)
    current_level_nids = set(n.id for n in initial_notes)
    
    for d in range(degree):
        log.append(f"\n--- Analyzing Degree {d+1} ---")
        next_level_nids = set()
        
        for nid in current_level_nids:
            note = mw.col.get_note(nid)
            note_label = note[SOURCE_LABEL_FIELD].split('(')[0].strip()
            connected = extract_connected_points(note[TARGET_FIELD])
            
            log.append(f"Base card: {note_label} ({len(connected)} links found)")
            
            for pt in connected:
                search_term = EXCEPTIONS.get(pt, pt)
                
                # Search using 'grammar' deck
                query = f'deck:grammar "{SOURCE_LABEL_FIELD}:{search_term}*"'
                found_nids = mw.col.find_notes(query)
                
                if not found_nids:
                    # Fallback wildcard search
                    query = f'deck:grammar "{SOURCE_LABEL_FIELD}:*{search_term}*"'
                    found_nids = mw.col.find_notes(query)
                
                if not found_nids:
                    log.append(f"  ⚠️ Nothing found for: {search_term}")
                    continue

                best_match_found = False
                for f_nid in found_nids:
                    f_note = mw.col.get_note(f_nid)
                    label_content = f_note[SOURCE_LABEL_FIELD].split('(')[0].strip()
                    clean_label = label_content.replace(" ", "").replace(" / ", "/")
                    
                    if clean_label == search_term or clean_label == pt:
                        if f_nid not in processed_nids:
                            next_level_nids.add(f_nid)
                            processed_nids.add(f_nid)
                            log.append(f"  ✅ Newly found: {label_content}")
                        best_match_found = True
                        break 
                
                if not best_match_found:
                    log.append(f"  ❌ No exact match for {search_term}")
        
        if not next_level_nids:
            log.append("No further connections found on this level.")
            break
        current_level_nids = next_level_nids

    # Tagging Execution (only if requested)
    if add_tag_opt and final_tag:
        for nid in processed_nids:
            note = mw.col.get_note(nid)
            note.add_tag(final_tag)
            mw.col.update_note(note)
        set_clipboard(final_tag)

    mw.progress.finish()
    
    # 1. Filtered Deck Execution
    if add_tag_opt and final_tag and create_filtered_opt:
        deck_name = f"Relation Review: {tag_prefix}"
        did = mw.col.decks.id(deck_name, type=1)
        deck = mw.col.decks.get(did)
        deck['terms'][0][0] = f'tag:"{final_tag}"'
        deck['resched'] = True
        mw.col.decks.save(deck)
        mw.col.sched.rebuild_filtered_deck(did)
        tooltip(f"Filtered Deck '{deck_name}' created.")

    # 2. Browser Overview Execution
    if show_in_browser_opt:
        if add_tag_opt and final_tag:
            # If tagged, search via tag (looks cleaner)
            browser.search_for(f'tag:"{final_tag}"')
        else:
            # If NOT tagged, search directly by note IDs
            if processed_nids:
                nid_query = " or ".join(f"nid:{nid}" for nid in processed_nids)
                browser.search_for(nid_query)
            else:
                tooltip("No cards found to display.")

    # 3. Final Report & Logs
    final_msg = f"Done!\n"
    if add_tag_opt and final_tag:
        final_msg += f"Tag: {final_tag}\n"
        final_msg += f"Cards tagged: {len(processed_nids)}"
    else:
        final_msg += f"Tagging disabled.\nCards found: {len(processed_nids)}"

    if show_log_opt:
        log_text = "\n".join(log)
        final_msg += f"\n\nSEARCH LOG:\n{log_text}"

    showInfo(final_msg)
    browser.model.reset()
    mw.reset()

from aqt import mw, gui_hooks
from aqt.utils import showInfo, qconnect
from aqt.qt import QAction

# Core logic for a single note
def process_note(note):
    cards = note.cards()
    if len(cards) < 2:
        return False

    # Sort by template index (0 = Card 1, 1 = Card 2)
    cards.sort(key=lambda x: x.ord)
    
    # Check status (queue -1 is suspended)
    c1_suspended = cards[0].queue == -1
    c2_suspended = cards[1].queue == -1

    new_value = ""
    # 1. suspended, 2. active -> black
    if c1_suspended and not c2_suspended:
        new_value = "black"
    # 1. active, 2. suspended -> white
    elif not c1_suspended and c2_suspended:
        new_value = "white"
    # 1. active, 2. active -> green
    elif not c1_suspended and not c2_suspended:
        new_value = "green"
    elif c1_suspended and c2_suspended:
        new_value = "yellow"

    if "SiblingSyncInfo" in note:
        if note["SiblingSyncInfo"] != new_value:
            note["SiblingSyncInfo"] = new_value
            note.flush()
            return True
    return False

# Function for the Tools menu (Global Scan)
def update_all_miningsimple():
    # Search for all notes of type 'miningsimple'
    note_ids = mw.col.find_notes("note:miningsimple")
    
    if not note_ids:
        showInfo("No notes of type 'miningsimple' found.")
        return

    changes = 0
    mw.checkpoint("Global Sibling Sync")
    
    for nid in note_ids:
        note = mw.col.get_note(nid)
        if process_note(note):
            changes += 1

    mw.reset()
    showInfo(f"Global scan completed: {changes} 'miningsimple' notes updated.")

# Function for the browser (selective selection)
def update_sibling_situation_browser(browser):
    selected_notes = browser.selected_notes()
    if not selected_notes:
        showInfo("Please select notes in the browser first.")
        return

    changes = 0
    mw.checkpoint("Update sibling situation")

    for nid in selected_notes:
        note = mw.col.get_note(nid)
        if process_note(note):
            changes += 1

    mw.reset()
    showInfo(f"Browser update completed: {changes} notes updated.")

# Menu registration
def setup_menus(browser=None):
    # If we are in the browser, add it to 'Edit'
    if browser:
        action = QAction("Update sibling situation", browser)
        qconnect(action.triggered, lambda: update_sibling_situation_browser(browser))
        browser.form.menuEdit.addAction(action)
    
    # Global Tools menu (called at startup)
    else:
        action = QAction("Sync all miningsimple notes", mw)
        qconnect(action.triggered, update_all_miningsimple)
        mw.form.menuTools.addAction(action)

# Register hooks
gui_hooks.browser_menus_did_init.append(setup_menus)

# For the main menu (Tools) we use a simple initialization
def on_main_window_init():
    setup_menus(browser=None)

# The main menu is populated directly when the add-on loads
on_main_window_init()
